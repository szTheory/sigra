```elixir
@doc """
Loads and classifies an invitation for the `InvitationAcceptLive` mount.

Returns a branch tuple the LV assigns directly. Does NOT run any Multi —
the caller invokes `accept/3` or `accept_with_signup/3` separately on
user action.
"""
@spec load_for_view(map(), String.t(), struct() | nil) ::
        {:signup, struct(), struct(), struct()}
        | {:accept, struct(), struct(), struct(), struct()}
        | {:mismatch, struct(), struct()}
        | {:invalid, :invalid}
        | {:expired, struct() | nil}
        | {:revoked, struct()}
        | {:already_accepted, struct(), boolean()}
def load_for_view(config, signed_token, current_user) do
  case verify_and_load(config, signed_token) do
    {:ok, invitation, _bound_email} ->
      org = config.repo.get!(config.schemas.organization, invitation.organization_id)
      inviter = config.repo.get!(config.schemas.user, invitation.invited_by_id)
      classify_for_view(invitation, org, inviter, current_user)

    {:error, :invalid} -> {:invalid, :invalid}
    {:error, :expired} -> {:expired, nil}
    {:error, :revoked} ->
      # Revoked rows still return from DB; re-fetch for display context
      # (accept_changeset with inviter name).
      {:revoked, nil}
    {:error, :already_accepted} -> {:already_accepted, nil, false}
  end
end

defp classify_for_view(invitation, org, inviter, nil),
  do: {:signup, invitation, org, inviter}
defp classify_for_view(invitation, org, inviter, %{email: email} = user) do
  if String.downcase(to_string(email)) == String.downcase(to_string(invitation.email)) do
    {:accept, invitation, org, inviter, user}
  else
    {:mismatch, invitation, user}
  end
end
```

  across 7 render branches.

  contains ZERO `phx-click`/`phx-submit` accept controls. Even if every
  server-side guard regressed, there is no accept form in the rendered

  defense locked in via UI-SPEC §Structural Invariants and enforced by
  plan-checker grep.

  use <%= app_module %>Web, :live_view

  # from the generated Sigra.Organizations wrapper

  @impl true
  def mount(%{"token" => signed_token}, _session, socket) do
    config = Organizations.__config__()
    current_user = socket.assigns[:current_user]

    case Sigra.Organizations.Invitations.load_for_view(config, signed_token, current_user) do
      {:signup, invitation, org, inviter} ->
        {:ok,
         socket
         |> assign(:branch, :signup)
         |> assign(:token, signed_token)
         |> assign(:invitation, invitation)
         |> assign(:organization, org)
         |> assign(:inviter, inviter)
         |> assign(:signup_form, to_form(%{"email" => invitation.email, "password" => "", "password_confirmation" => ""}, as: :user))
         |> assign(:page_title, "Join #{org.name}")}

      {:accept, invitation, org, inviter, user} ->
        {:ok,
         socket
         |> assign(:branch, :accept)
         |> assign(:token, signed_token)
         |> assign(:invitation, invitation)
         |> assign(:organization, org)
         |> assign(:inviter, inviter)
         |> assign(:current_user, user)
         |> assign(:page_title, "Join #{org.name}")}

      {:mismatch, invitation, user} ->
        {:ok,
         socket
         |> assign(:branch, :mismatch)
         |> assign(:invitation, invitation)
         |> assign(:current_user, user)
         |> assign(:page_title, "Invitation not for you")}

      {:invalid, :invalid} ->
        {:ok, socket |> assign(:branch, :invalid) |> assign(:page_title, "Invalid invitation")}

      {:expired, invitation} ->
        {:ok, socket |> assign(:branch, :expired) |> assign(:invitation, invitation)
              |> assign(:page_title, "Invitation expired")}

      {:revoked, invitation} ->
        {:ok, socket |> assign(:branch, :revoked) |> assign(:invitation, invitation)
              |> assign(:page_title, "Invitation no longer valid")}

      {:already_accepted, invitation, _is_member} ->
        {:ok, socket |> assign(:branch, :already_accepted) |> assign(:invitation, invitation)
              |> assign(:page_title, "Already accepted")}

  @impl true
  def handle_event("accept_invitation", _params, %{assigns: %{branch: :accept}} = socket) do
    token = socket.assigns.token
    user = socket.assigns.current_user
    org = socket.assigns.organization

    case Organizations.accept_invitation(token, user) do
      {:ok, _result} ->
        {:noreply,
         socket
         |> put_flash(:info, "You joined #{org.name}.")
         |> redirect(to: ~p"/organizations/#{org.slug}")}

      {:error, :already_accepted} ->
        {:noreply, socket |> put_flash(:info, "You're already a member of #{org.name}.") |> redirect(to: ~p"/organizations/#{org.slug}")}

      {:error, reason} ->
        {:noreply, put_flash(socket, :error, accept_error_copy(reason))}
    end
  end

  # :mismatch, this clause does not match, and there is NO fallback clause
  # that handles the event. Crash-safe by design — the mismatch branch
  # renders no such button, so this event can never originate from a
  # legitimate client. Test 17 (Jetstream #907 regression) synthesizes the
  # event via render_click and asserts ArgumentError is raised.

  def handle_event("accept_with_signup", %{"user" => user_params}, %{assigns: %{branch: :signup}} = socket) do
    token = socket.assigns.token
    org = socket.assigns.organization

        {:noreply,
         socket

      {:error, %Ecto.Changeset{} = cs} ->
        {:noreply, assign(socket, :signup_form, to_form(cs, as: :user))}

      {:error, reason} ->
        {:noreply, put_flash(socket, :error, accept_error_copy(reason))}
    end
  end
