**4. Add the `handle_event("invite_member", ...)` handler:**

```elixir
@impl true
def handle_event("invite_member", %{"invitation" => params}, socket) do
  scope = socket.assigns.current_scope
  org = socket.assigns.active_organization

  attrs = %{
    actor: scope,
    organization_id: org.id,
    email: params["email"],
    role: safe_role(params["role"]),
    invited_by_id: scope.user.id
  }

  case create_invitation(attrs) do
    {:ok, invitation} ->
      {:noreply,
       socket
       |> put_flash(:info, "Invitation sent to #{invitation.email}.")
       |> stream_insert(:pending_invitations, invitation, at: 0)
       |> assign(:invite_form, to_form(%{"email" => "", "role" => "member"}, as: :invitation))
       |> push_event("js-exec", %{to: "#invite-member-modal", attr: "data-close"})}

    {:error, %Ecto.Changeset{} = cs} ->
      {:noreply, assign(socket, :invite_form, to_form(cs, as: :invitation))}

    {:error, :already_member} ->
      form =
        to_form(
          %{"email" => params["email"], "role" => params["role"]},
          as: :invitation,
          errors: [email: {"#{params["email"]} is already a member of this organization.", []}]
        )
      {:noreply, assign(socket, :invite_form, form)}

    {:error, :rate_limited_user} ->
      {:noreply,
       socket
       |> put_flash(:error, "You have invited too many people today. Try again tomorrow.")
       |> push_event("js-exec", %{to: "#invite-member-modal", attr: "data-close"})}

    {:error, :rate_limited_org} ->
      {:noreply,
       socket
       |> put_flash(:error, "This organization has reached its daily invitation limit. Try again tomorrow.")
       |> push_event("js-exec", %{to: "#invite-member-modal", attr: "data-close"})}

    {:error, :unauthorized} ->
      {:noreply, put_flash(socket, :error, "You do not have permission to invite members.")}
  end
end

defp safe_role("owner"), do: :owner
defp safe_role("admin"), do: :admin
defp safe_role(_), do: :member
```

**4. Add the revoke handlers:**

```elixir
@impl true
def handle_event("open_revoke_modal", %{"id" => id}, socket) do
  case find_pending(socket, id) do
    nil ->
      {:noreply, put_flash(socket, :error, "Invitation not found.")}

    invitation ->
      {:noreply,
       socket
       |> assign(:revoking_invitation, invitation)
       |> push_event("js-exec", %{to: "#revoke-invitation-modal", attr: "data-open"})}
  end
end

def handle_event("cancel_revoke", _, socket) do
  {:noreply,
   socket
   |> assign(:revoking_invitation, nil)
   |> push_event("js-exec", %{to: "#revoke-invitation-modal", attr: "data-close"})}
end

def handle_event("confirm_revoke", %{"id" => id}, socket) do
  case revoke_invitation(id, socket.assigns.current_scope) do
    {:ok, invitation} ->
      {:noreply,
       socket
       |> put_flash(:info, "Invitation for #{invitation.email} revoked.")
       |> stream_delete(:pending_invitations, invitation)
       |> update(:pending_count, &max(&1 - 1, 0))
       |> assign(:revoking_invitation, nil)
       |> push_event("js-exec", %{to: "#revoke-invitation-modal", attr: "data-close"})}

    {:error, :not_pending} ->
      {:noreply,
       socket
       |> put_flash(:error, "This invitation has already been accepted and cannot be revoked.")
       |> assign(:revoking_invitation, nil)
       |> refresh_pending_invitations()
       |> push_event("js-exec", %{to: "#revoke-invitation-modal", attr: "data-close"})}

    {:error, :unauthorized} ->
      {:noreply, put_flash(socket, :error, "You do not have permission to revoke invitations.")}

    {:error, :not_found} ->
      {:noreply, put_flash(socket, :error, "Invitation not found.")}
  end
end

defp find_pending(socket, id) do
  org = socket.assigns.active_organization
  Enum.find(list_pending_invitations(org), fn i -> to_string(i.id) == to_string(id) end)
end

defp refresh_pending_invitations(socket) do
  assign_pending_invitations(socket)
end
```
