<threat_model>

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| HTTP request → library | `signed_token` from URL is attacker-controllable; HMAC verify is the first gate |
| current_user → library | `current_user.email` is trusted (came from authenticated session); invitation.email is trusted (stored in citext column written by admin) |
| user_params → library | `user_params.email` in `accept_with_signup/3` is attacker-controllable; MUST be rejected if != invitation.email even though the UI locks the field |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-17-02 | Spoofing + Tampering | `verify_and_load/2` HMAC verify | mitigate | Calls `Sigra.Token.verify_invite_envelope/3` which rejects any byte-level tamper before DB touch. Returns `{:error, :invalid}` (generic — no info leak about what differed). |
| T-17-02 (DB re-check) | Defense-in-depth | `verify_and_load/2` bound_email check | mitigate | After DB lookup by `hashed_token`, re-asserts `bound_email == downcase(db_row.email)`. If HMAC passed but the hashed_token somehow maps to a row with a different email (shouldn't be possible given Plan 17-02's unique index on hashed_token), returns `{:error, :invalid}` uniformly. |

| T-17-08 | Replay (accept twice) | `accept/3` + `accept_with_signup/3` state guards | mitigate | `verify_and_load/2` checks `accepted_at == nil AND revoked_at == nil AND expires_at > now()` in the same query or as a post-lookup guard. This means the invitee arrives confirmed — no separate email confirmation round-trip — because HMAC-bound invite acceptance is proof of email ownership. |

```elixir
  # ---------- verify_and_load/2 (private helper) ----------

  @spec verify_and_load(map(), String.t()) ::
          {:ok, struct(), String.t()}
          | {:error, :invalid | :expired | :revoked | :already_accepted}
  defp verify_and_load(config, signed_token) do
    ttl_seconds = div(config.invitation_ttl, 1_000)
    schema = config.schemas.invitation

    with {:ok, envelope} <-
           Sigra.Token.verify_invite_envelope(config.secret_key_base, signed_token, ttl_seconds),
         %{} = inv <- config.repo.get_by(schema, hashed_token: envelope.hashed_token),
         :ok <- assert_bound_email(envelope.bound_email, inv.email),
         :ok <- assert_pending_state(inv) do
      {:ok, inv, envelope.bound_email}
    else
      nil -> {:error, :invalid}
      {:error, :invalid} -> {:error, :invalid}
      {:error, :expired} -> {:error, :expired}
      {:error, :revoked} -> {:error, :revoked}
      {:error, :already_accepted} -> {:error, :already_accepted}
      {:error, :email_mismatch} -> {:error, :invalid}  # collapse to :invalid — no info leak
    end
  end

  defp assert_bound_email(bound_email, row_email) do
    if String.downcase(bound_email) == String.downcase(to_string(row_email)) do
      :ok
    else
      {:error, :email_mismatch}
    end
  end

  defp assert_pending_state(%{accepted_at: nil, revoked_at: nil, expires_at: exp}) do
    if DateTime.compare(exp, DateTime.utc_now()) == :gt do
      :ok
    else
      {:error, :expired}
    end
  end
  defp assert_pending_state(%{accepted_at: t}) when not is_nil(t), do: {:error, :already_accepted}
  defp assert_pending_state(%{revoked_at: t}) when not is_nil(t), do: {:error, :revoked}

  # ---------- accept/3 (signed-in-match path, INV-06) ----------

  @spec accept(map(), String.t(), struct()) ::
          {:ok, %{membership: struct(), invitation: struct()}}
          | {:error, :invalid | :expired | :revoked | :already_accepted | :mismatch}
  def accept(config, signed_token, %{email: user_email} = current_user) do
    with {:ok, invitation, _bound_email} <- verify_and_load(config, signed_token),
         :ok <- assert_user_matches_invitation(current_user, invitation),
         {:ok, org} <- fetch_org(config, invitation),
         {:ok, result} <- run_accept_multi(config, invitation, current_user, org) do
      {:ok, result}
    end
  end

  defp assert_user_matches_invitation(%{email: user_email}, %{email: inv_email}) do
    if String.downcase(to_string(user_email)) == String.downcase(to_string(inv_email)) do
      :ok
    else
      {:error, :mismatch}
    end
  end

  defp fetch_org(config, invitation) do
    case config.repo.get(config.schemas.organization, invitation.organization_id) do
      nil -> {:error, :invalid}
      org -> {:ok, org}
    end
  end

  defp run_accept_multi(config, invitation, user, org) do
    now = DateTime.utc_now() |> DateTime.truncate(:second)
    user_scope = %{user: user, membership: nil, active_organization: org}

    accept_changeset =
      invitation
      |> Ecto.Changeset.change(%{accepted_at: now, accepted_by_id: user.id})

    Multi.new()
    |> Multi.append(Sigra.Organizations.add_member_multi(config, user_scope, org, user, invitation.role))
    |> Multi.update(:accept_invitation, accept_changeset)
    |> append_audit(config, "organization.invitation.accepted", user_scope,
         metadata: %{invitation_id: invitation.id, role: to_string(invitation.role)})
    |> config.repo.transact()
    |> case do
         {:ok, %{membership: m, accept_invitation: inv}} ->
           {:ok, %{membership: m, invitation: inv}}
         {:error, _step, reason, _} ->
           {:error, reason}
       end
  end
```

```elixir
  # ---------- accept_with_signup/3 (anonymous signup path, INV-05) ----------

  @spec accept_with_signup(map(), String.t(), map()) ::
          {:ok, %{user: struct(), membership: struct(), invitation: struct()}}
          | {:error,
             :invalid | :expired | :revoked | :already_accepted | :email_mismatch
             | Ecto.Changeset.t()}
  def accept_with_signup(config, signed_token, user_params) do
    with {:ok, invitation, _bound_email} <- verify_and_load(config, signed_token),
         :ok <- assert_signup_email_matches(user_params, invitation.email),
         {:ok, org} <- fetch_org(config, invitation),
         {:ok, result} <- run_accept_with_signup_multi(config, invitation, user_params, org) do
      {:ok, result}
    end
  end

  defp assert_signup_email_matches(user_params, inv_email) do
    posted_email = Map.get(user_params, "email") || Map.get(user_params, :email) || ""
    if String.downcase(to_string(posted_email)) == String.downcase(to_string(inv_email)) do
      :ok
    else
      {:error, :email_mismatch}
    end
  end

  defp run_accept_with_signup_multi(config, invitation, user_params, org) do
    now = DateTime.utc_now() |> DateTime.truncate(:second)
    # Force email to the invitation email — belt-and-suspenders even though
    # assert_signup_email_matches/2 already guarded against mismatch.
    params_with_locked_email =
      user_params
      |> Map.new(fn {k, v} -> {to_string(k), v} end)
      |> Map.put("email", invitation.email)

    auth_opts = [changeset_fn: config.user_registration_changeset_fn]

    accept_stamp_fn = fn %{confirm_user: user} ->
      Ecto.Changeset.change(invitation, %{
        accepted_at: now,
        accepted_by_id: user.id
      })
    end

    Multi.new()
    |> Multi.append(Sigra.Auth.register_user_multi(params_with_locked_email, auth_opts))
    |> Multi.run(:confirm_user, fn repo, %{user: user} ->
         repo.update(Ecto.Changeset.change(user, confirmed_at: now))
       end)
    |> Multi.append(
         Sigra.Organizations.add_member_multi(
           config,
           %{user: nil, membership: nil, active_organization: org},
           org,
           {:changes_key, :confirm_user},
           invitation.role
         )
       )
    |> Multi.update(:accept_invitation, accept_stamp_fn)
    |> append_audit(config, "organization.invitation.accepted",
         %{user: nil, membership: nil, active_organization: org},
         metadata: %{invitation_id: invitation.id, role: to_string(invitation.role), path: "signup"})
    |> config.repo.transact()
    |> case do
         {:ok, %{confirm_user: user, membership: m, accept_invitation: inv}} ->
           {:ok, %{user: user, membership: m, invitation: inv}}
         {:error, :user, %Ecto.Changeset{} = cs, _} ->
           {:error, cs}
         {:error, _step, reason, _} ->
           {:error, reason}
       end
  end
```
