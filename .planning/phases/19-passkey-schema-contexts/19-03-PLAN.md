## Sign-count policy machine (D-05)

```elixir
defmodule Sigra.Passkeys.SignCountPolicy do
  @moduledoc """
  Pure sign-count regression policy machine (D-05, PK-08).

  Implements three modes — `:warn`, `:require_reauth`, `:revoke` — with the
  W3C-spec-mandated zero-to-zero carve-out (PITFALLS.md §P-4) for sync'd
  passkeys (Apple iCloud Keychain, Google Password Manager) that legitimately
  return `sign_count = 0` on every assertion.

  This module is PURE — no DB, no wax_, no audit. The caller
  (`Sigra.Passkeys.Authentication`) interprets the `{:regression, policy}`
  return and dispatches side effects.
  """

  @type policy :: :warn | :require_reauth | :revoke
  @type result :: :ok | {:regression, policy()}

  @spec evaluate(non_neg_integer(), non_neg_integer(), policy()) :: result()
  def evaluate(stored, presented, policy)
      when is_integer(stored) and stored >= 0
      and is_integer(presented) and presented >= 0
      and policy in [:warn, :require_reauth, :revoke] do
    cond do
      stored == 0 and presented == 0 -> :ok                     # P-4 carve-out
      presented > stored              -> :ok                     # monotonic increase
      true                            -> {:regression, policy}
    end
  end
end
```

## Sigra.Passkeys.Authentication (D-07 StrongKey guard)

```elixir
defmodule Sigra.Passkeys.Authentication do
  @moduledoc """
  WebAuthn authentication ceremony primitive wrapping `Wax.authenticate/6`.

  Security invariants owned here:
  - D-07 (PK-07 / StrongKey CVE-2025-26788): pre-lookup `{user_id, credential_id}`
    BEFORE calling wax_. A mismatch returns `{:error, :credential_not_owned}`
    without any wax_ call made.
  - W3C WebAuthn §7.2 step 6.3.2: `userHandle`, when present in the assertion,
    must equal `user.id` — additional belt-and-braces on the StrongKey guard.
  - D-05 (PK-08): sign-count regression dispatched to `SignCountPolicy.evaluate/3`.
  - D-11 (PK-04 / P-3): advisory `rp_id` mismatch check — log but do not block.

  Plug.Conn-free (D-09). Challenge is an explicit argument.
  """

  alias Sigra.Passkeys.{CoseKey, SignCountPolicy}

  @spec new_challenge(Sigra.Config.t(), keyword()) :: Wax.Challenge.t()
  def new_challenge(%Sigra.Config{} = config, _opts \\ []) do
    Wax.new_authentication_challenge(
      origin: config.passkeys[:origin],
      rp_id: config.passkeys[:rp_id],
      user_verification: config.passkeys[:user_verification] || :preferred,
      timeout: config.passkeys[:timeout_ms] || 60_000
    )
  end

  @type assertion :: %{
          required(:credential_id) => binary(),
          required(:authenticator_data) => binary(),
          required(:signature) => binary(),
          required(:client_data_json) => binary(),
          required(:challenge) => Wax.Challenge.t(),
          optional(:user_handle) => binary() | nil
        }

  @spec verify(Sigra.Config.t(), user :: map(), assertion(), keyword()) ::
          {:ok, row :: struct(), auth_data :: Wax.AuthenticatorData.t()}
          | {:error, :credential_not_owned}
          | {:error, atom()}
  def verify(%Sigra.Config{} = config, user, assertion, opts \\ []) do
    schema = Keyword.fetch!(opts, :user_passkey_schema)
    credential_id = Map.fetch!(assertion, :credential_id)

    # STEP 1 — StrongKey guard: DB pre-lookup scoped to user_id AND credential_id
    # This is the PK-07 defense. Do not call Wax.authenticate/6 until this succeeds.
    case config.repo.get_by(schema, user_id: user.id, credential_id: credential_id) do
      nil ->
        {:error, :credential_not_owned}

      %{} = row ->
        # STEP 2 — userHandle check (W3C §7.2 step 6.3.2)
        case Map.get(assertion, :user_handle) do
          nil ->
            do_verify(config, row, assertion, credential_id)

          uh when uh == to_string(user.id) or uh == user.id ->
            do_verify(config, row, assertion, credential_id)

          _mismatch ->
            {:error, :credential_not_owned}
        end
    end
  end

  defp do_verify(config, row, assertion, credential_id) do
    # STEP 3 — advisory rp_id drift check (D-11)
    stored_rp_id = row.rp_id
    current_rp_id = config.passkeys[:rp_id]

    if stored_rp_id && current_rp_id && stored_rp_id != current_rp_id do
      require Logger
      Logger.warning(
        "passkey rp_id drift: stored=#{inspect(stored_rp_id)} current=#{inspect(current_rp_id)} credential_id=#{Base.url_encode64(credential_id, padding: false)}"
      )
    end

    # STEP 4 — decrypt + deserialize (Cloak handles decrypt on load)
    cose_key = CoseKey.deserialize(row.public_key)

    # STEP 5 — build single-element allow_credentials (username-first mode, D-07)
    credentials = [{row.credential_id, cose_key}]

    # STEP 6 — call wax_
    case Wax.authenticate(
           credential_id,
           Map.fetch!(assertion, :authenticator_data),
           Map.fetch!(assertion, :signature),
           Map.fetch!(assertion, :client_data_json),
           Map.fetch!(assertion, :challenge),
           credentials
         ) do
      {:ok, auth_data} ->
        {:ok, row, auth_data}

      {:error, reason} ->
        {:error, {:wax_authenticate_failed, reason}}
    end
  end
end
```

## Sigra.Passkeys.authenticate/4 (public D-08 surface)

```elixir
@spec authenticate(Sigra.Config.t(), user :: map(), assertion :: map(), keyword()) ::
        {:ok, Sigra.Passkeys.Credential.t()}
        | {:error, :credential_not_owned}       # D-07 StrongKey guard fired
        | {:error, :sign_count_regression}      # :require_reauth or :revoke
        | {:error, atom()}
def authenticate(%Sigra.Config{} = config, user, assertion, opts \\ []) do
  Sigra.Telemetry.span([:sigra, :passkeys, :authenticate], %{user_id: user.id}, fn ->
    validated = NimbleOptions.validate!(opts, @authenticate_opts_schema)
    schema = resolve_user_passkey_schema!(config, opts)
    policy = Keyword.get(validated, :sign_count_policy) || config.passkeys[:sign_count_policy] || :warn

    verify_opts = [user_passkey_schema: schema]

    with {:ok, row, auth_data} <- Authentication.verify(config, user, assertion, verify_opts),
         {:ok, credential}      <- handle_sign_count_and_update(config, row, auth_data, policy) do
      {:ok, credential}
    end
  end)
end
```

## handle_sign_count_and_update/4 (atomic DB update + audit)

```elixir
defp handle_sign_count_and_update(config, row, auth_data, policy) do
  stored = row.sign_count
  presented = auth_data.sign_count

  case SignCountPolicy.evaluate(stored, presented, policy) do
    :ok ->
      update_sign_count_ok(config, row, presented)

    {:regression, :warn} ->
      update_with_regression_audit(config, row, presented, :warn, :continue)

    {:regression, :require_reauth} ->
      update_with_regression_audit(config, row, presented, :require_reauth, :block)

    {:regression, :revoke} ->
      revoke_credential_with_audit(config, row, presented)
  end
end
```
