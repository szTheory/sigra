## Repo.transact flow (D-12, D-06, D-11)

```elixir
def register(%Sigra.Config{} = config, user, params, opts \\ []) do
  Sigra.Telemetry.span([:sigra, :passkeys, :register], %{user_id: user.id}, fn ->
    validated = NimbleOptions.validate!(opts, @register_opts_schema)
    cap = Keyword.get(validated, :max_per_user, config.passkeys[:max_per_user] || 10)
    schema = resolve_user_passkey_schema!(config, opts)

    config.repo.transact(fn ->
      with {:ok, :cap_ok}       <- check_cap(config, user, schema, cap),
           {:ok, extracted}     <- Registration.verify(config, user, params),
           attrs                 = build_attrs(user, extracted, config, params),
           multi = Ecto.Multi.new()
                   |> Ecto.Multi.insert(:passkey, schema.create_changeset(struct(schema), attrs))
                   |> Sigra.Audit.log_multi_safe(
                        "passkey.register.success",
                        passkey_audit_opts(config) ++ [
                          actor_id: user.id,
                          metadata: %{
                            credential_id: Base.url_encode64(attrs.credential_id, padding: false),
                            rp_id: attrs.rp_id,
                            aaguid: attrs.aaguid
                          }
                        ]
                      ),
           {:ok, %{passkey: row}} <- config.repo.transaction(multi) do
        {:ok, Sigra.Passkeys.Credential.from_schema(row)}
      else
        {:error, :cap_reached, info} -> {:error, :passkey_cap_reached, info}
        {:error, %Ecto.Changeset{} = cs} -> {:error, cs}
        {:error, other} -> {:error, other}
      end
    end)
  end)
end

defp check_cap(config, user, schema, cap) do
  count = config.repo.aggregate(from(p in schema, where: p.user_id == ^user.id), :count, :id)

  if count >= cap do
    {:error, :cap_reached, %{count: count, cap: cap}}
  else
    {:ok, :cap_ok}
  end
end

defp build_attrs(user, %{credential_id: cid, cose_key: cose, aaguid: aa, sign_count: sc}, config, params) do
  %{
    user_id: user.id,
    credential_id: cid,
    public_key: Sigra.Passkeys.CoseKey.serialize(cose),
    sign_count: sc,
    aaguid: cast_aaguid(aa),                     # nil-safe
    nickname: Map.get(params, :nickname),
    device_hint: Map.get(params, :device_hint),
    transports: Map.get(params, :transports, []),
    rp_id: config.passkeys[:rp_id]               # D-11: always from config, never from client
  }
end

defp cast_aaguid(nil), do: nil
defp cast_aaguid(bin) when is_binary(bin) and byte_size(bin) == 16 do
  case Ecto.UUID.cast(bin) do
    {:ok, uuid} -> uuid
    :error -> nil
  end
end
```

## Sigra.Passkeys.Registration.verify/3 shape

```elixir
defmodule Sigra.Passkeys.Registration do
  @moduledoc """
  WebAuthn registration ceremony primitive wrapping `Wax.register/3`.

  Plug.Conn-free (D-09). Caller (Phase 20's `Sigra.Plug.PasskeyChallenge`)
  owns challenge storage and lookup.
  """

  @spec new_challenge(Sigra.Config.t(), keyword()) :: Wax.Challenge.t()
  def new_challenge(%Sigra.Config{} = config, _opts \\ []) do
    Wax.new_registration_challenge(
      origin: config.passkeys[:origin],
      rp_id: config.passkeys[:rp_id],
      user_verification: config.passkeys[:user_verification] || :preferred,
      attestation: config.passkeys[:attestation] || :none,
      timeout: config.passkeys[:timeout_ms] || 60_000
    )
  end

  @spec verify(Sigra.Config.t(), user :: map(), map()) ::
          {:ok, %{credential_id: binary(), cose_key: map(), aaguid: binary() | nil, sign_count: non_neg_integer()}}
          | {:error, atom()}
  def verify(%Sigra.Config{}, _user, %{attestation_object: ao, client_data_json: cdj, challenge: %Wax.Challenge{} = ch}) do
    case Wax.register(ao, cdj, ch) do
      {:ok, {auth_data, _attestation_result}} ->
        acd = auth_data.attested_credential_data
        {:ok, %{
          credential_id: acd.credential_id,
          cose_key: acd.credential_public_key,
          aaguid: Wax.AuthenticatorData.get_aaguid(auth_data),
          sign_count: auth_data.sign_count
        }}

      {:error, reason} ->
        {:error, {:wax_register_failed, reason}}
    end
  end
end
```
