```elixir
defmodule Sigra.Install.Features.Passkeys do
  @moduledoc """
  `Sigra.Install.Feature` implementation for the passkeys feature.

  Owns every template under `priv/templates/sigra.install/passkeys/` and the
  single migration that creates the `user_passkeys` table.

  `enabled?/1` checks `Keyword.get(opts, :passkeys, false)`. Passkeys are
  opt-in for v1.1 — the default-on bit is locked at Phase 22 (GEN-related),
  NOT here.

  ## Isolation invariant (Pitfall X-3)

  This module contains ZERO references to `Features.Core`, `Features.Organizations`,
  or `Features.Admin`. Cross-feature coupling is prohibited.
  """

  @behaviour Sigra.Install.Feature

  alias Sigra.Install.Injection

  @impl true
  def enabled?(opts), do: Keyword.get(opts, :passkeys, false)

  @impl true
  def files(binding) do
    otp_app = Keyword.fetch!(binding, :otp_app) |> to_string()
    context_slug = binding |> Keyword.get(:context_alias, "Accounts") |> Macro.underscore()

    [
      {:eex, "passkeys/user_passkey.ex",
       Path.join(["lib", otp_app, context_slug, "user_passkey.ex"])}
    ]
  end

  @impl true
  def migrations(_binding) do
    [
      {:user_passkeys, "passkeys/create_user_passkeys.exs", "create_user_passkeys.exs"}
    ]
  end

  @impl true
  def injections(_binding), do: []
end
```

## Vault template (D-13 promotion — verbatim from priv/templates/sigra.gen.oauth/vault.ex)

```eex
defmodule <%= app_module %>.Vault do
  @moduledoc """
  Cloak vault for encrypting sensitive fields at rest.

  Uses AES-256-GCM encryption with a key derived from the `CLOAK_KEY`
  environment variable.

  ## Generating a key

      iex> 32 |> :crypto.strong_rand_bytes() |> Base.encode64()

  Set the result as the `CLOAK_KEY` env var in runtime.exs.

  ## Consumers

  - `<%= context_module %>.UserPasskey.public_key` (Phase 19, PK-03)
  - `<%= context_module %>.UserMFACredential.encrypted_secret` (Phase 6, MFA)
  - `<%= context_module %>.UserIdentity.access_token` + `refresh_token` (Phase 5, OAuth)

  Generated by `mix sigra.install` when any of `--passkeys`, `--mfa`, or
  `--oauth` is enabled.
  """

  use Cloak.Vault, otp_app: <%= inspect(otp_app) %>

  @impl GenServer
  def init(config) do
    config =
      Keyword.put(config, :ciphers,
        default: {
          Cloak.Ciphers.AES.GCM,
          tag: "AES.GCM.V1",
          key: decode_env!("CLOAK_KEY"),
          iv_length: 12
        }
      )

    {:ok, config}
  end

  defp decode_env!(var) do
    var |> System.fetch_env!() |> Base.decode64!()
  end
end
```

```elixir
defp promote_vault(upgrade_ctx) do
  cond do
    not any_sensitive_feature_enabled?(upgrade_ctx) ->
      :skip

    already_promoted?(upgrade_ctx) ->
      :skip

    true ->
      with :ok <- write_vault_template(upgrade_ctx),

           :ok <- inject_vault_supervision_child(upgrade_ctx) do
        print_cloak_key_banner()
        :ok
      end
  end

defp inject_vault_supervision_child(%{application_ex_path: path, app_module: app_module}) do
  contents = File.read!(path)

  case Sigra.Install.Injector.inject_vault_child(contents, app_module) do
    {:ok, new} -> File.write!(path, new); :ok
    {:already_injected, _} -> :ok
  end
end
```

## D-15 boot-check (Sigra.Application)

```elixir
# lib/sigra/application.ex — add to start/2 callback

def start(_type, _args) do
  verify_vault_or_raise!()
  # ...existing children supervision...
end

@doc false
def verify_vault_or_raise!() do
  # Only check when a config explicitly enables passkeys at runtime.
  # Hostless mode (Sigra itself running no host app) has no vault at all.

  passkeys_enabled =
    Application.get_env(:sigra, :passkeys, [])
    |> Keyword.get(:enabled, false)

  app_module = Application.get_env(:sigra, :app_module)
  encrypted_binary_module = app_module && Module.concat([app_module, Encrypted, Binary])

  cond do
    not passkeys_enabled -> :ok
    is_nil(encrypted_binary_module) -> :ok
    not Code.ensure_loaded?(encrypted_binary_module) -> :ok
    stub?(encrypted_binary_module) ->
      raise """
      Sigra: passkeys are enabled but #{inspect(encrypted_binary_module)} is the
      passthrough stub. public_key would be stored in plaintext.

      Run `mix sigra.upgrade` to promote the Cloak vault, or add
      `{:cloak_ecto, "~> 1.3"}` to your deps and follow the upgrade guide at
      guides/upgrading-to-v1.1.md#enabling-passkeys.
      """
    true -> :ok
  end
end

defp stub?(module) do
  # Stub: use Ecto.Type (passthrough). Real: use Cloak.Ecto.Binary.
  # Detect via behaviour / exported functions.
  functions = module.__info__(:functions)

  # Real Cloak.Ecto.Binary exports :cast/1, :dump/1, :load/1 AND the Cloak
  # callbacks. The stub is a straight Ecto.Type passthrough — no Cloak-specific
  # callbacks.
  not function_exported?(module, :before_encrypt, 1)
end
```
