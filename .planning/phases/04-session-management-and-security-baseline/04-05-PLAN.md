```elixir
create table(:user_sessions) do
  add :user_id, references(:users, type: :binary_id, on_delete: :delete_all), null: false
  add :hashed_token, :binary, null: false
  add :type, :string, null: false, default: "standard"
  add :ip, :string
  add :user_agent, :text
  add :geo_city, :string
  add :geo_country_code, :string, size: 2
  add :last_active_at, :utc_datetime_usec, null: false
  add :sudo_at, :utc_datetime_usec

  timestamps(type: :utc_datetime_usec, updated_at: false)
end

create unique_index(:user_sessions, [:hashed_token])
create index(:user_sessions, [:user_id])
create index(:user_sessions, [:user_id, :type])
create index(:user_sessions, [:inserted_at])
```

Per D-63: IP stored as `:string` (not `:inet`) for MySQL/SQLite portability. Timestamps as `:utc_datetime_usec`.
Per D-06: Indexes on hashed_token (unique), user_id, (user_id, type), inserted_at.

Note: If the migration template uses EEx bindings (like `<%= otp_app %>`), follow the existing pattern. The user_id reference type should match the existing users table primary key type.

**Sudo controller** (`priv/templates/sigra.install/sudo_controller.ex`) per D-20 to D-23, UI-SPEC Surface 2:

```elixir
defmodule <%= inspect context.web_module %>.Auth.SudoController do
  use <%= inspect context.web_module %>, :controller

  def new(conn, _params) do
    render(conn, :new, return_to: conn.params["return_to"] || ~p"/")
  end

  def create(conn, %{"sudo" => %{"password" => password, "return_to" => return_to}}) do
    user = conn.assigns.current_scope.user
    config = <%= inspect context.module %>.Auth.sigra_config()

    case Sigra.Crypto.verify_password(password, user.hashed_password) do
      true ->
        session = conn.private[:sigra_session]
        <%= inspect context.module %>.Auth.confirm_sudo(session.hashed_token)

        conn
        |> put_flash(:info, "Password confirmed.")
        |> redirect(to: return_to || ~p"/")

      false ->
        conn
        |> put_flash(:error, "Incorrect password. Please try again.")
        |> render(:new, return_to: return_to)
    end
  end
end
```

  ip = Map.get(details, :ip, "Unknown")

  |> text_body(suspicious_login_text(ip, location, device, time))
