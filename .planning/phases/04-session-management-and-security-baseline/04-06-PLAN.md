```elixir
defmodule <%= inspect context.web_module %>.Auth.SessionLive do
  use <%= inspect context.web_module %>, :live_view

  def mount(_params, _session, socket) do
    user = socket.assigns.current_scope.user
    sessions = <%= inspect context.module %>.Auth.list_sessions(user)
    current_token = get_connect_params(socket)["_sigra_token"]

    {:ok, assign(socket,
      sessions: sessions,
      current_token: current_token,
      page_title: "Active Sessions"
    )}
  end

  def render(assigns) do
    ~H"""
    <div class="mx-auto max-w-2xl">
      <.header>
        Active Sessions
        <:subtitle>These devices are currently signed in to your account.</:subtitle>
      </.header>

      <div class="mt-8 space-y-4">
        <div :for={session <- @sessions} class="flex items-start justify-between p-4 bg-gray-50 rounded-lg border border-gray-200">
          <div class="flex-1">
            <div class="flex items-center gap-2">
              <span class="text-sm font-semibold">
                <%= device_name(session) %>
              </span>
              <span
                :if={current_session?(session, @current_token)}
                class="inline-flex items-center rounded-md bg-brand/10 px-2 py-1 text-xs font-semibold text-brand"
              >
                This device
              </span>
            </div>
            <div class="mt-1 text-sm text-gray-500">
              <span><%= session.ip || "Unknown IP" %></span>
              <span class="mx-1">&middot;</span>
              <span><%= location(session) %></span>
            </div>
            <div class="mt-1 text-sm text-gray-500">
              <%= relative_time(session.last_active_at) %>
            </div>
          </div>
          <div>
            <%= if current_session?(session, @current_token) do %>
              <.button
                phx-click="revoke_current"
                phx-value-token={Base.url_encode64(session.hashed_token)}
                data-confirm="This is your current session. Revoking it will log you out. Continue?"
                class="text-sm text-red-600 bg-red-50 hover:bg-red-100"
              >
                Revoke session
              </.button>
            <% else %>
              <.button
                phx-click="revoke"
                phx-value-token={Base.url_encode64(session.hashed_token)}
                class="text-sm text-red-600 bg-red-50 hover:bg-red-100"
              >
                Revoke session
              </.button>
            <% end %>
          </div>
        </div>
      </div>

      <div :if={length(@sessions) > 1} class="mt-8 border-t border-gray-200 pt-6">
        <.button
          phx-click="revoke_all"
          data-confirm="This will end all sessions including your current one. You will be logged out."
          class="text-red-600 bg-red-50 hover:bg-red-100"
        >
          Log out of all devices
        </.button>
      </div>
    </div>
    """
  end

  def handle_event("revoke", %{"token" => encoded_token}, socket) do
    hashed_token = Base.url_decode64!(encoded_token)
    <%= inspect context.module %>.Auth.revoke_session(hashed_token)

    sessions = <%= inspect context.module %>.Auth.list_sessions(socket.assigns.current_scope.user)
    {:noreply, socket |> put_flash(:info, "Session revoked.") |> assign(sessions: sessions)}
  end

  def handle_event("revoke_current", %{"token" => _encoded_token}, socket) do
    user = socket.assigns.current_scope.user
    <%= inspect context.module %>.Auth.revoke_all_sessions(user, except_token: nil)

    {:noreply, redirect(socket, to: ~p"/users/log_in")}
  end

  def handle_event("revoke_all", _params, socket) do
    user = socket.assigns.current_scope.user
    <%= inspect context.module %>.Auth.revoke_all_sessions(user)

    {:noreply, redirect(socket, to: ~p"/users/log_in")}
  end

  defp device_name(%{parsed_ua: %{browser: browser, browser_version: version, os: os}})
       when browser != "Unknown" do
    if version, do: "#{browser} #{version} on #{os}", else: "#{browser} on #{os}"
  end
  defp device_name(_), do: "Unknown device"

  defp location(%{geo_city: city, geo_country_code: country}) when is_binary(city),
    do: "#{city}, #{country}"
  defp location(_), do: "Unknown location"

  defp relative_time(nil), do: "Never"
  defp relative_time(datetime) do
    seconds = DateTime.diff(DateTime.utc_now(), datetime, :second)
    cond do
      seconds < 60 -> "Active just now"
      seconds < 3600 -> "Active #{div(seconds, 60)} minutes ago"
      seconds < 86400 -> "Active #{div(seconds, 3600)} hours ago"
      true -> "Active #{div(seconds, 86400)} days ago"
    end
  end

  defp current_session?(%{hashed_token: token}, current_token) when is_binary(current_token),
    do: token == Base.url_decode64!(current_token)
  defp current_session?(_, _), do: false
end
```
