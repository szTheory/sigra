```elixir
defmodule Sigra.Install.PurelyAdditiveTest do
  use ExUnit.Case, async: false

  @moduletag :purely_additive

  alias Sigra.Install.Injection
  alias Sigra.Install.Runner

  defmodule FakeFeature do
    @behaviour Sigra.Install.Feature

    @impl true
    def enabled?(_opts), do: true

    @impl true
    def files(binding) do
      tmp = Keyword.fetch!(binding, :tmp_dir)
      [{:eex, "fake/hello.txt", Path.join(tmp, "hello.txt")}]
    end

    @impl true
    def injections(binding) do
      tmp = Keyword.fetch!(binding, :tmp_dir)

      [
        %Injection{
          target: Path.join(tmp, "router.ex"),
          marker: "# fake:injected",
          anchor: :at_top,
          content: "# fake:injected\n# fake line"
        }
      ]
    end

    @impl true
    def migrations(_binding) do
      [{:fake_slot, "fake/fake_migration.exs", "create_fake.exs"}]
    end

    @impl true
    def post_instructions(_binding, _report), do: ["Fake feature done.\n"]
  end

  setup do
    tmp = Path.join(System.tmp_dir!(), "sigra_purely_additive_#{System.unique_integer([:positive])}")
    File.mkdir_p!(Path.join(tmp, "priv/repo/migrations"))
    File.write!(Path.join(tmp, "router.ex"), "defmodule Router do\nend\n")

    # Write a fake template that Runner.find_template/1 can resolve via the host-app override path.
    File.mkdir_p!(Path.join(tmp, "priv/templates/sigra.install/fake"))
    File.write!(Path.join(tmp, "priv/templates/sigra.install/fake/hello.txt"), "hello world\n")
    File.write!(
      Path.join(tmp, "priv/templates/sigra.install/fake/fake_migration.exs"),
      "defmodule Repo.Migrations.CreateFake do\nend\n"
    )

    on_exit(fn -> File.rm_rf!(tmp) end)
    {:ok, tmp: tmp}
  end

  test "FakeFeature is walked by Runner with zero edits to runner.ex or sigra.install.ex", %{tmp: tmp} do
    # Runner must be called with a feature list — no special-casing per feature.
    cwd_before = File.cwd!()
    File.cd!(tmp)

    try do
      binding = [tmp_dir: tmp, opts: []]
      assert {:ok, _report, _instrs} = Runner.run([FakeFeature], binding, [])

      # File written
      assert File.read!(Path.join(tmp, "hello.txt")) =~ "hello world"

      # Injection applied
      assert File.read!(Path.join(tmp, "router.ex")) =~ "# fake:injected"

      # Migration written with a timestamped filename
      migrations = File.ls!(Path.join(tmp, "priv/repo/migrations"))
      assert Enum.any?(migrations, fn f -> f =~ ~r/^\d{14}_create_fake\.exs$/ end)
    after
      File.cd!(cwd_before)
    end
  end

  test "sigra.install.ex contains no feature-specific branches (grep assertion)" do
    source = File.read!("lib/mix/tasks/sigra.install.ex")

    # Forbidden patterns — any presence means the walker has feature-specific logic.
    refute source =~ "Features.Organizations", "sigra.install.ex must not mention Features.Organizations"
    refute source =~ "Features.Passkeys", "sigra.install.ex must not mention Features.Passkeys"
    refute source =~ "Features.Admin", "sigra.install.ex must not mention Features.Admin"

    # The walker should reference the feature list via @features, not case-match on feature modules.
    assert source =~ "@features", "sigra.install.ex must declare @features module attribute"
    refute source =~ ~r/case\s+feature\s+do/, "sigra.install.ex must not case-match on feature modules"
  end

  test "runner.ex contains no feature-specific branches" do
    source = File.read!("lib/sigra/install/runner.ex")

    refute source =~ "Features.Core", "runner.ex must be feature-agnostic"
    refute source =~ "Features.Organizations"
    refute source =~ "Features.Passkeys"
    refute source =~ "Features.Admin"
  end
end
```

```elixir
defmodule Sigra.Install.IsolationTest do
  use ExUnit.Case, async: true

  @moduletag :isolation

  @forbidden_symbols [
    "Features.Organizations",
    "Features.Passkeys",
    "Features.Admin",
    "OrganizationMembership",
    "OrganizationInvitation",
    "UserPasskey",
    "Sigra.Passkeys",
    "Sigra.Organizations"
  ]

  test "lib/sigra/install/features/core.ex has no forbidden future-feature references" do
    source = File.read!("lib/sigra/install/features/core.ex")

    Enum.each(@forbidden_symbols, fn symbol ->
      refute source =~ symbol,
             "Features.Core contains forbidden reference to #{symbol} — " <>
               "Core must not know about other features (Pitfall X-1)"
    end)
  end

  test "every template under priv/templates/sigra.install/core/ has no forbidden future-feature references" do
    "priv/templates/sigra.install/core"
    |> File.ls!()
    |> Enum.each(fn filename ->
      path = Path.join("priv/templates/sigra.install/core", filename)
      content = File.read!(path)

      Enum.each(@forbidden_symbols, fn symbol ->
        refute content =~ symbol,
               "Template #{filename} contains forbidden reference to #{symbol} — " <>
                 "core/ templates must compile with no other features enabled (Pitfall X-3)"
      end)
    end)
  end

  test "there are exactly 45 templates under priv/templates/sigra.install/core/" do
    files = File.ls!("priv/templates/sigra.install/core")
    assert length(files) == 45
  end
end
```
