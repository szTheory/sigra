defmodule Sigra.Install.Runner do
  @moduledoc """
  Generic walker over a `[Sigra.Install.Feature]` list. Feature-agnostic:
  adding `Features.Organizations`, `Features.Passkeys`, or `Features.Admin`
  in a later phase requires ZERO edits to this module — only a new entry
  in the caller's feature list.

  See Phase 11 CONTEXT.md D-03 (full decomposition) and the "purely
  additive" invariant V-PA-01.
  """

  alias Sigra.Install.{Injector, MigrationTimestamps, Report}

  @spec run(features :: [module()], binding :: keyword(), opts :: keyword()) ::
          {:ok, Report.t(), post_instructions :: iodata()}
  def run(features, binding, opts) when is_list(features) do
    active = Enum.filter(features, fn f -> f.enabled?(opts) end)
    timestamps = MigrationTimestamps.allocate(active, DateTime.utc_now())

    {report, instructions} =
      Enum.reduce(active, {Report.new(), []}, fn feature, {r, instrs} ->
        r1 = run_files(feature, binding, r)
        r2 = run_injections(feature, binding, r1)
        r3 = run_migrations(feature, binding, r2, timestamps[feature] || %{})
        feature_instrs = feature.post_instructions(binding, r3)
        {r3, [instrs, feature_instrs]}
      end)

    {:ok, report, instructions}
  end

  defp run_files(feature, binding, report) do
    Enum.reduce(feature.files(binding), report, fn {:eex, source, target}, r ->
      if File.exists?(target) do
        Report.record_skipped(r, target, "already exists")
      else
        content = EEx.eval_file(find_template(source), binding)
        File.mkdir_p!(Path.dirname(target))
        File.write!(target, content)
        Report.record_generated(r, target)
      end
    end)
  end

  defp run_injections(feature, binding, report) do
    Enum.reduce(feature.injections(binding), report, fn injection, r ->
      case Injector.apply(injection) do
        {:ok, :injected} -> Report.record_modified(r, injection.target)
        {:ok, :already_present} -> Report.record_skipped(r, injection.target, "already injected")
        {:error, reason} -> Report.record_manual_action(r, "Injection #{injection.marker} failed: #{inspect(reason)}")
      end
    end)
  end

  defp run_migrations(feature, binding, report, slot_timestamps) do
    Enum.reduce(feature.migrations(binding), report, fn {slot, source, basename}, r ->
      migrations_dir = "priv/repo/migrations"
      File.mkdir_p!(migrations_dir)

      # Re-run idempotency: if a file with a matching basename already exists
      # (after stripping any 14-digit prefix), reuse it — don't allocate a new timestamp.
      existing =
        migrations_dir
        |> File.ls!()
        |> Enum.find(fn f -> strip_timestamp(f) == basename end)

      case existing do
        nil ->
          ts = Map.fetch!(slot_timestamps, slot)
          target = Path.join(migrations_dir, "#{ts}_#{basename}")
          content = EEx.eval_file(find_template(source), binding)
          File.write!(target, content)
          Report.record_generated(r, target)

        filename ->
          Report.record_skipped(r, Path.join(migrations_dir, filename), "migration already exists")
      end
    end)
  end

  defp strip_timestamp(filename) do
    String.replace(filename, ~r/^\d{14}_/, "")
  end

  defp find_template(source) do
    # Host-app override first
    override = Path.join([File.cwd!(), "priv/templates/sigra.install", source])

    if File.exists?(override) do
      override
    else
      Application.app_dir(:sigra, "priv/templates/sigra.install/#{source}")
    end
  end
end

defmodule Mix.Tasks.Sigra.Install do
  @moduledoc """
  Generates Sigra authentication scaffold.

  Creates migrations, schemas, context module, plug helpers,
  and optional LiveView pages for authentication.

  ## Usage

      mix sigra.install Accounts User users

  The first argument is the context name (e.g., `Accounts`),
  the second is the schema name (e.g., `User`), and the third
  is the table name (e.g., `users`).

  ## Options

    * `--live` / `--no-live` - Generate LiveView pages (default: true)
    * `--binary-id` / `--no-binary-id` - Use UUID primary keys (default: true)
    * `--table` - Override the table name
    * `--api` - Generate API token controller
    * `--jwt` - Generate JWT controller
    * `--yes` - Non-interactive mode (no prompts)
  """
  @shortdoc "Generates Sigra authentication scaffold"

  use Mix.Task

  alias Sigra.Install.{Report, Runner}

  @features [Sigra.Install.Features.Core]

  @switches [
    live: :boolean,
    binary_id: :boolean,
    table: :string,
    api: :boolean,
    jwt: :boolean,
    yes: :boolean
  ]

  @default_opts [live: true, api: false, jwt: false, binary_id: true]

  @impl true
  def run(args) do
    {opts, parsed, _} = OptionParser.parse(args, switches: @switches)
    opts = Keyword.merge(@default_opts, opts)

    case parsed do
      [context_name, schema_name, table_name] ->
        validate_args!(context_name, schema_name, table_name)
        binding = build_binding(context_name, schema_name, opts[:table] || table_name, opts)
        {:ok, report, instructions} = Runner.run(@features, binding, opts)
        print_summary(report, instructions)

      _ ->
        Mix.raise("""
        Expected exactly 3 arguments: context_name schema_name table_name

        Usage:
            mix sigra.install Accounts User users
            mix sigra.install Accounts User users --no-live
        """)
    end
  end

  defp validate_args!(context_name, schema_name, table_name) do
    unless context_name =~ ~r/^[A-Z][A-Za-z0-9]*(\.[A-Z][A-Za-z0-9]*)*$/ do
      Mix.raise("The context name must be a valid Elixir module name, got: #{context_name}")
    end

    unless schema_name =~ ~r/^[A-Z][A-Za-z0-9]*$/ do
      Mix.raise("The schema name must be a valid Elixir module name, got: #{schema_name}")
    end

    unless table_name =~ ~r/^[a-z][a-z0-9_]*$/ do
      Mix.raise("The table name must be a valid identifier, got: #{table_name}")
    end
  end

  defp build_binding(context_name, schema_name, table_name, opts) do
    base = Mix.Phoenix.base()
    web_module = Module.concat([Mix.Phoenix.web_module(base)])
    otp_app = Mix.Phoenix.otp_app()

    [
      context_module: inspect(Module.concat([base, context_name])),
      context_alias: context_name,
      schema_module: inspect(Module.concat([base, context_name, schema_name])),
      schema_alias: schema_name,
      table_name: table_name,
      web_module: inspect(web_module),
      web_path: Macro.underscore(inspect(web_module)),
      otp_app: to_string(otp_app),
      base: inspect(base),
      from_email: "noreply@example.com",
      log_in_url: "/users/log_in",
      opts: opts
    ]
  end

  defp print_summary(report, instructions) do
    Mix.shell().info(IO.iodata_to_binary(Report.render_summary(report)))
    Mix.shell().info(IO.iodata_to_binary(instructions))
  end
end

defmodule Sigra.Install.IdempotencyTest do
  use ExUnit.Case, async: false

  @moduletag :integration
  @moduletag :idempotency
  @moduletag timeout: 180_000

  alias Sigra.InstallFixture

  setup_all do
    {:ok, app_dir} = InstallFixture.generate_tmp_app("idempotency_app")
    {:ok, first_output} = InstallFixture.run_sigra_install(app_dir, ["Accounts", "User", "users", "--yes"])
    %{app_dir: app_dir, first_output: first_output}
  end

  test "second invocation produces zero new file writes and zero new injections", %{app_dir: app_dir} do
    before = InstallFixture.snapshot_tree(app_dir)
    mtimes_before = collect_mtimes(app_dir)

    {:ok, second_output} = InstallFixture.run_sigra_install(app_dir, ["Accounts", "User", "users", "--yes"])

    after_snap = InstallFixture.snapshot_tree(app_dir)
    mtimes_after = collect_mtimes(app_dir)

    # Byte-identity
    assert InstallFixture.diff_trees(before, after_snap) == :ok,
           "second install run changed file contents — GEN-04 broken"

    # Modification times stable (stronger check: no files were rewritten even with the same content)
    assert mtimes_before == mtimes_after,
           "second install run touched file mtimes — runner is not skipping existing files properly"

    # Summary text should show skipped entries
    assert second_output =~ "Skipped",
           "second install should print a Skipped column populated"

    assert second_output =~ "already exists" or second_output =~ "already injected",
           "second install should indicate skip reasons"
  end

  test "first-run report contains generated entries, second-run report contains skipped entries", %{
    first_output: first_output,
    app_dir: app_dir
  } do
    # First run output sanity check
    assert first_output =~ "Generated"

    # Capture second run output
    {:ok, second_output} = InstallFixture.run_sigra_install(app_dir, ["Accounts", "User", "users", "--yes"])

    assert second_output =~ "Skipped"
  end

  defp collect_mtimes(dir) do
    dir
    |> Path.join("**/*")
    |> Path.wildcard()
    |> Enum.filter(&File.regular?/1)
    |> Enum.reject(fn p -> p =~ "_build/" or p =~ "deps/" or p =~ ".git/" end)
    |> Map.new(fn p ->
      %File.Stat{mtime: mtime} = File.stat!(p)
      {p, mtime}
    end)
  end
end
