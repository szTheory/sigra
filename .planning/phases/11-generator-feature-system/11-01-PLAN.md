1. `generate_tmp_app(app_name :: String.t()) :: {:ok, Path.t()} | {:error, term()}` —
   - Creates a unique tmp dir under `System.tmp_dir!()` keyed on `:os.system_time(:nanosecond)` to avoid collisions.
   - Shells out via `System.cmd("mix", ["phx.new", app_name, "--no-assets", "--no-mailer", "--no-install", "--database", "postgres"], cd: tmp_dir, stderr_to_stdout: true)`.
   - Returns `{:ok, Path.join(tmp_dir, app_name)}` on exit code 0, `{:error, output}` otherwise.
   - Mix.shell().info a yellow "* generating phx.new fixture at \#{path}" line so CI logs show progress.

2. `run_sigra_install(app_dir :: Path.t(), args :: [String.t()]) :: {:ok, output :: String.t()} | {:error, output :: String.t()}` —
   - Shells out via `System.cmd("mix", ["sigra.install" | args], cd: app_dir, env: [{"MIX_ENV", "test"}], stderr_to_stdout: true)`.
   - Canonical test args: `["Accounts", "User", "users", "--yes"]`.
   - Returns `{:ok, output}` on exit code 0, `{:error, output}` otherwise.
   - The returned `output` is the captured stdout+stderr blob — **callers MUST normalize before diffing** via `normalize_stdout/2` (below). This output is what the GEN-05 4-column summary regression test compares against `test/fixtures/install_golden/STDOUT.txt`.

6. `normalize_stdout(output :: String.t(), app_dir :: Path.t()) :: String.t()` —
   - Strips ANSI escape codes: `String.replace(output, ~r/\e\[[0-9;]*m/, "")`.
   - Strips absolute tmp paths by replacing `app_dir` with the literal `"<APP_DIR>"`.
   - Strips migration timestamp prefixes by replacing `~r/\d{14}_/` with `"<TS>_"`.
   - Returns the normalized string suitable for byte-comparison against the committed `STDOUT.txt` fixture.

7. `diff_stdout(expected :: String.t(), actual :: String.t()) :: :ok | {:error, myers}` —
   - Returns `:ok` if `expected == actual`.
   - On mismatch, returns `{:error, String.myers_difference(expected, actual)}` for unified-diff-style failure rendering by the harness.

3. `snapshot_tree(app_dir :: Path.t()) :: %{Path.t() => binary()}` —
   - Walks `app_dir` recursively using `Path.wildcard(Path.join(app_dir, "**/*"))`.
   - Filters out `_build/`, `deps/`, `.git/`, `assets/node_modules/`, `priv/static/`.
   - Returns a map of `relative_path => File.read!(abs_path)`.
   - The map keys must be project-relative (no absolute prefix) and normalized with `normalize_migration_filenames/1` applied to each key.

4. `normalize_migration_filenames(paths :: [Path.t()] | Path.t()) :: [Path.t()] | Path.t()` —
   - Replaces any path segment matching `~r/^\d{14}_/` with `"TIMESTAMP_"` (keeping the rest of the basename).
   - Rule cited from CONTEXT.md D-05. Example: `priv/repo/migrations/20260411120000_create_users.exs` becomes `priv/repo/migrations/TIMESTAMP_create_users.exs`.
   - Accepts either a list or a single path; returns the same shape.

5. `diff_trees(expected :: %{Path.t() => binary()}, actual :: %{Path.t() => binary()}) :: :ok | {:error, [diff_entry]}` —
   - Returns `:ok` if key sets and all byte values match.
   - On mismatch, builds a list of diff entries: `{:missing, path}`, `{:extra, path}`, `{:content_diff, path, myers}` where `myers = String.myers_difference(expected_bytes, actual_bytes)`.
   - The walker MUST sort keys so diff output is deterministic.

defmodule Sigra.Install.GoldenDiffTest do
  use ExUnit.Case, async: false

  @moduletag :integration
  @moduletag :golden_diff
  @moduletag timeout: 180_000

  alias Sigra.InstallFixture

  @golden_dir "test/fixtures/install_golden"

  setup_all do
    {:ok, app_dir} = InstallFixture.generate_tmp_app("golden_diff_app")
    {:ok, raw_stdout} = InstallFixture.run_sigra_install(app_dir, ["Accounts", "User", "users", "--yes"])
    normalized_stdout = InstallFixture.normalize_stdout(raw_stdout, app_dir)
    %{app_dir: app_dir, snapshot: InstallFixture.snapshot_tree(app_dir), stdout: normalized_stdout}
  end

  test "generated tree is byte-identical to committed golden snapshot (migration filenames normalized)",
       %{snapshot: actual} do
    expected =
      case File.ls(@golden_dir) do
        {:ok, []} ->
          :capture
        {:error, :enoent} ->
          :capture
        {:ok, _files} ->
          load_golden(@golden_dir)
      end

    case expected do
      :capture ->
        capture_snapshot!(@golden_dir, actual)
        flunk("""
        Golden fixture empty. Captured #{map_size(actual)} files into #{@golden_dir}.
        Re-run the test to verify diff-ability. If the fixture was intentionally cleared,
        follow the regeneration runbook in 11-01-PLAN.md and commit the result.
        """)

      expected_map ->
        case InstallFixture.diff_trees(expected_map, actual) do
          :ok -> :ok
          {:error, diffs} -> flunk(format_diffs(diffs))
        end
    end
  end

  test "captured installer stdout matches committed STDOUT.txt fixture (GEN-05 byte gate)",
       %{stdout: actual_stdout} do
    stdout_path = Path.join(@golden_dir, "STDOUT.txt")

    case File.read(stdout_path) do
      {:error, :enoent} ->
        File.mkdir_p!(@golden_dir)
        File.write!(stdout_path, actual_stdout)
        flunk("""
        STDOUT.txt fixture was missing; captured #{byte_size(actual_stdout)} bytes into #{stdout_path}.
        Re-run the test to verify diff-ability. If the fixture was intentionally cleared,
        follow the regeneration runbook in 11-01-PLAN.md.
        """)

      {:ok, expected_stdout} ->
        case InstallFixture.diff_stdout(expected_stdout, actual_stdout) do
          :ok ->
            :ok

          {:error, myers} ->
            rendered =
              myers
              |> Enum.map_join("", fn
                {:eq, s} -> s
                {:del, s} -> "[-" <> s <> "-]"
                {:ins, s} -> "[+" <> s <> "+]"
              end)
              |> String.slice(0, 4000)

            flunk("""
            Installer stdout drifted from STDOUT.txt fixture (GEN-05 regression):

            #{rendered}

            If the change is intentional, regenerate the fixture per the runbook in 11-01-PLAN.md.
            """)
        end
    end
  end

  test "re-running installer on the same app produces zero new/modified files", %{app_dir: app_dir} do
    before = InstallFixture.snapshot_tree(app_dir)
    {:ok, _out} = InstallFixture.run_sigra_install(app_dir, ["Accounts", "User", "users", "--yes"])
    after_snap = InstallFixture.snapshot_tree(app_dir)

    assert InstallFixture.diff_trees(before, after_snap) == :ok,
           "re-running mix sigra.install should be idempotent (GEN-04 baseline)"
  end

  # --- helpers ---

  defp load_golden(dir) do
    dir
    |> Path.join("**/*")
    |> Path.wildcard()
    |> Enum.filter(&File.regular?/1)
    |> Map.new(fn abs ->
      rel = Path.relative_to(abs, dir)
      {rel, File.read!(abs)}
    end)
  end

  defp capture_snapshot!(dir, snapshot) do
    File.mkdir_p!(dir)
    Enum.each(snapshot, fn {rel, bytes} ->
      abs = Path.join(dir, rel)
      File.mkdir_p!(Path.dirname(abs))
      File.write!(abs, bytes)
    end)
  end

  defp format_diffs(diffs) do
    diffs
    |> Enum.take(20)
    |> Enum.map_join("\n\n", fn
      {:missing, p} -> "MISSING in generated tree: #{p}"
      {:extra, p} -> "EXTRA in generated tree: #{p}"
      {:content_diff, p, myers} ->
        "CONTENT DIFF at #{p}:\n" <>
          (myers
           |> Enum.map_join("", fn
             {:eq, s} -> s
             {:del, s} -> "[-" <> s <> "-]"
             {:ins, s} -> "[+" <> s <> "+]"
           end)
           |> String.slice(0, 2000))
    end)
  end
end
