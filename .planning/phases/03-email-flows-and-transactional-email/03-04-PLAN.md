**Create `priv/templates/sigra.install/reset_password_controller.ex` (per D-31, D-44):**

Controller handles 4 routes:
- `GET /users/reset-password` -- renders request form (enter email)
- `POST /users/reset-password` -- processes request, sends email (enumeration-safe)
- `GET /users/reset-password/:token` -- renders new password form
- `PUT /users/reset-password/:token` -- processes password change

```elixir
defmodule <%= web_module %>.ResetPasswordController do
  use <%= web_module %>, :controller

  alias <%= context_module %>

  # GET /users/reset-password -- request form
  def new(conn, _params) do
    render(conn, :new)
  end

  # POST /users/reset-password -- send reset email
  def create(conn, %{"user" => %{"email" => email}}) do
    # D-38: Always show same message regardless of email existence
    <%= context_module %>.deliver_user_reset_password_instructions(
      email,
      &url(~p"/users/reset-password/#{&1}")
    )

    conn
    |> put_flash(:info, dgettext("sigra", "If your email is in our system, you will receive reset instructions shortly."))
    |> redirect(to: ~p"/users/log-in")
  end

  # GET /users/reset-password/:token -- new password form
  def edit(conn, %{"token" => token}) do
    case <%= context_module %>.get_user_by_reset_password_token(token) do
      nil ->
        # D-30: expired/invalid token page
        render(conn, :expired)

      user ->
        changeset = <%= context_module %>.change_user_password(user)
        render(conn, :edit, changeset: changeset, token: token, user: user)
    end
  end

  # PUT /users/reset-password/:token -- change password
  def update(conn, %{"token" => token, "user" => password_params}) do
    case <%= context_module %>.get_user_by_reset_password_token(token) do
      nil ->
        render(conn, :expired)

      user ->
        case <%= context_module %>.reset_user_password(user, password_params) do
          {:ok, user} ->
            # D-29: auto-login after reset
            conn
            |> put_flash(:info, dgettext("sigra", "Your password has been reset."))
            |> <%= web_module %>.UserAuth.log_in_user(user)

          {:error, changeset} ->
            render(conn, :edit, changeset: changeset, token: token, user: user)
        end
    end
  end
end
```

**Extend `priv/templates/sigra.install/user_auth.ex` (per D-02):**

Add a new plug function `require_confirmed_user/2` that checks `user.confirmed_at`:

```elixir
@doc """
Plug that handles unconfirmed users based on configuration.

When `:unconfirmed_access` is `:allow_with_banner` (default):
  - Sets info flash: "Please confirm your email. Check your inbox or request a new confirmation email."
  - Allows request to continue

When `:unconfirmed_access` is `:block`:
  - Sets error flash: "You must confirm your email before logging in. We've sent a new confirmation email."
  - Auto-resends confirmation email (D-04)
  - Redirects to confirmation page
"""
def require_confirmed_user(conn, opts \\ []) do
  user = conn.assigns[:current_scope] && conn.assigns.current_scope.user

  cond do
    is_nil(user) ->
      conn

    user.confirmed_at != nil ->
      conn

    unconfirmed_access_mode(opts) == :allow_with_banner ->
      conn
      |> put_flash(:info, dgettext("sigra", "Please confirm your email. Check your inbox or request a new confirmation email."))

    unconfirmed_access_mode(opts) == :block ->
      # D-04: auto-resend confirmation on blocked login attempt
      <%= context_module %>.deliver_user_confirmation_instructions(user, &url(~p"/users/confirm/#{&1}"))

      conn
      |> put_flash(:error, dgettext("sigra", "You must confirm your email before logging in. We've sent a new confirmation email."))
      |> redirect(to: ~p"/users/confirm")
      |> halt()
  end
end

defp unconfirmed_access_mode(opts) do
  Keyword.get(opts, :unconfirmed_access, :allow_with_banner)
end
```

Also update the existing `log_in_user/3` function to handle the unconfirmed flow:
- After login, if user is unconfirmed and mode is `:block`, redirect to `/users/confirm` instead of home
- If mode is `:allow_with_banner`, proceed normally but the `require_confirmed_user` plug adds the banner flash

The plug is designed to be placed in the router pipeline after `fetch_current_scope_for_user`:
