```elixir
defmodule <%= context_module %>.Mailer do
  @moduledoc """
  Mailer implementation for Sigra authentication emails.

  Implements the `Sigra.Mailer` behaviour by building a `Swoosh.Email`
  struct and delivering via the host app's Swoosh mailer.
  """
  @behaviour Sigra.Mailer

  @impl Sigra.Mailer
  def deliver(to, subject, body) do
    email = build_email(to, subject, body)

    case <%= app_module %>.Mailer.deliver(email) do
      {:ok, _metadata} = ok -> ok
      {:error, _reason} = error -> error
    end
  end

  defp build_email(to, subject, body) when is_binary(body) do
    import Swoosh.Email

    new()
    |> to(to)
    |> from({"<%= app_name %>", "<%= from_email %>"})
    |> subject(subject)
    |> text_body(body)
  end

  defp build_email(to, subject, %{html: html, text: text}) do
    import Swoosh.Email

    new()
    |> to(to)
    |> from({"<%= app_name %>", "<%= from_email %>"})
    |> subject(subject)
    |> html_body(html)
    |> text_body(text)
  end

  defp build_email(to, subject, %{text: text}) do
    build_email(to, subject, text)
  end
end
```

**Create `priv/templates/sigra.install/confirmation_controller.ex` (per D-06, D-43):**

Controller handles 4 routes:
- `GET /users/confirm` -- renders code entry form (D-06)
- `POST /users/confirm` -- processes code submission (D-06)
- `GET /users/confirm/:token` -- auto-confirms via link click (D-06)
- `POST /users/confirm/resend` -- resends confirmation email (D-04)

```elixir
defmodule <%= web_module %>.ConfirmationController do
  use <%= web_module %>, :controller

  alias <%= context_module %>

  def new(conn, _params) do
    render(conn, :new)
  end

  def create(conn, %{"code" => code}) do
    user = conn.assigns.current_scope.user

    case <%= context_module %>.confirm_user_by_code(user, code) do
      {:ok, _user} ->
        conn
        |> put_flash(:info, dgettext("sigra", "Your email has been confirmed."))
        |> redirect(to: ~p"/")

      {:error, :invalid_code} ->
        conn
        |> put_flash(:error, dgettext("sigra", "Invalid confirmation code. Please try again."))
        |> render(:new)

      {:error, :rate_limited} ->
        conn
        |> put_flash(:error, dgettext("sigra", "Too many attempts. Please wait a few minutes before trying again."))
        |> render(:new)

      {:error, :already_confirmed} ->
        conn
        |> put_flash(:info, dgettext("sigra", "Your email is already confirmed."))
        |> redirect(to: ~p"/")
    end
  end

  def confirm(conn, %{"token" => token}) do
    case <%= context_module %>.confirm_user(token) do
      {:ok, _user} ->
        conn
        |> put_flash(:info, dgettext("sigra", "Your email has been confirmed."))
        |> redirect(to: ~p"/")

      {:error, :already_confirmed} ->
        # D-09: friendly "already confirmed" page
        render(conn, :already_confirmed)

      {:error, :token_expired} ->
        # D-07: expired token page with resend button
        render(conn, :expired)

      {:error, :token_invalid} ->
        conn
        |> put_flash(:error, dgettext("sigra", "This confirmation link is invalid or has expired."))
        |> redirect(to: ~p"/users/confirm")
    end
  end

  def resend(conn, _params) do
    user = conn.assigns.current_scope.user

    case <%= context_module %>.deliver_user_confirmation_instructions(user, &url(~p"/users/confirm/#{&1}")) do
      {:ok, _} ->
        conn
        |> put_flash(:info, dgettext("sigra", "A new confirmation email has been sent."))
        |> redirect(to: ~p"/users/confirm")

      {:error, :already_confirmed} ->
        conn
        |> put_flash(:info, dgettext("sigra", "Your email is already confirmed."))
        |> redirect(to: ~p"/")

      {:error, :rate_limited} ->
        conn
        |> put_flash(:error, dgettext("sigra", "Please wait a few minutes before requesting another email."))
        |> redirect(to: ~p"/users/confirm")
    end
  end
end
```
