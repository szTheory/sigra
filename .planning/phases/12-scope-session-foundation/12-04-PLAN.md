@impl true
def fetch(hashed_token, opts) do
  repo = Keyword.fetch!(opts, :repo)
  schema = Keyword.fetch!(opts, :session_schema)
  case repo.get_by(schema, hashed_token: hashed_token) do
    nil -> {:error, :not_found}
    record -> {:ok, to_session(record)}
  end
end
