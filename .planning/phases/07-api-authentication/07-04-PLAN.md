    **1. Create priv/templates/sigra.install/api_token_controller.ex** (per D-57):

    JSON API controller for token management. Generated into host app.

    Endpoints per D-57:
    - `GET /api/tokens` -- list active tokens (paginated)
    - `POST /api/tokens` -- create token (requires sudo)
    - `DELETE /api/tokens/:id` -- revoke single token
    - `DELETE /api/tokens` -- revoke all tokens

    Protected by RequireAuthenticated + RequireScopes (api_tokens:read/write).

    ```elixir
    defmodule <%= @web_module %>.APITokenController do
      use <%= @web_module %>, :controller

      alias <%= @app_module %>.Auth

      def index(conn, params) do
        config = conn.assigns.sigra_config
        user = conn.assigns.current_scope
        opts = [cursor: params["cursor"], limit: parse_limit(params["limit"])]
        {tokens, next_cursor} = Auth.list_api_tokens(config, user.id, opts)

        json(conn, %{
          data: Enum.map(tokens, &token_json/1),
          next_cursor: next_cursor
        })
      end

      def create(conn, %{"token" => token_params}) do
        config = conn.assigns.sigra_config
        user = conn.assigns.current_scope
        attrs = %{
          name: token_params["name"],
          scopes: token_params["scopes"],
          expires_at: parse_expires_at(token_params["expires_at"])
        }

        case Auth.create_api_token(config, user, attrs) do
          {:ok, raw_key, token} ->
            conn
            |> put_status(:created)
            |> json(%{data: Map.merge(token_json(token), %{raw_key: raw_key})})

          {:error, changeset} when is_struct(changeset, Ecto.Changeset) ->
            conn
            |> put_status(:unprocessable_entity)
            |> json(%{error: "validation_error", details: changeset_errors(changeset)})

          {:error, reason} ->
            conn
            |> put_status(:unprocessable_entity)
            |> json(%{error: to_string(reason)})
        end
      end

      def delete(conn, %{"id" => id}) do
        config = conn.assigns.sigra_config
        case Auth.revoke_api_token(config, id) do
          {:ok, _token} -> json(conn, %{ok: true})
          {:error, :not_found} ->
            conn |> put_status(:not_found) |> json(%{error: "not_found"})
        end
      end

      def delete_all(conn, _params) do
        config = conn.assigns.sigra_config
        user = conn.assigns.current_scope
        {:ok, count} = Auth.revoke_all_api_tokens(config, user)
        json(conn, %{ok: true, revoked_count: count})
      end

      # ... private helpers: token_json/1, parse_limit/1, parse_expires_at/1, changeset_errors/1
    end
    ```

    **2. Create priv/templates/sigra.install/token_controller.ex** (per D-41, D-47):

    JWT authentication endpoints. Only generated with `--jwt` flag.

    Endpoints:
    - `POST /api/auth/token` -- authenticate (email+password), returns JWT or mfa_required
    - `POST /api/auth/token/refresh` -- refresh tokens
    - `POST /api/auth/token/mfa` -- complete MFA challenge for JWT
    - `DELETE /api/auth/token` -- revoke refresh token

    ```elixir
    defmodule <%= @web_module %>.TokenController do
      use <%= @web_module %>, :controller

      alias <%= @app_module %>.Auth

      def create(conn, %{"email" => email, "password" => password, "scopes" => scopes}) do
        config = conn.assigns.sigra_config
        case Auth.authenticate(config, %{"email" => email, "password" => password}) do
          {:ok, user} ->
            if user_has_mfa?(user) do
              # D-47: MFA required for JWT login
              mfa_token = generate_mfa_token(config, user)
              conn
              |> put_status(403)
              |> json(%{mfa_required: true, mfa_token: mfa_token})
            else
              case Auth.generate_jwt_tokens(config, user, scopes) do
                {:ok, tokens} -> json(conn, %{data: tokens})
                {:error, reason} ->
                  conn |> put_status(422) |> json(%{error: to_string(reason)})
              end
            end

          {:error, _reason} ->
            conn |> put_status(401) |> json(%{error: "invalid_credentials"})
        end
      end

      def refresh(conn, %{"refresh_token" => refresh_token}) do
        config = conn.assigns.sigra_config
        case Auth.refresh_jwt(config, refresh_token) do
          {:ok, tokens} -> json(conn, %{data: tokens})
          {:error, :reuse_detected} ->
            conn |> put_status(401) |> json(%{error: "token_revoked", detail: "Possible token theft detected. All tokens in this family have been revoked."})
          {:error, reason} ->
            conn |> put_status(401) |> json(%{error: to_string(reason)})
        end
      end

      def mfa(conn, %{"mfa_token" => mfa_token, "code" => code, "scopes" => scopes}) do

        # Implementation delegates to Auth MFA and JWT modules
      end

      def revoke(conn, %{"refresh_token" => refresh_token}) do

        Auth.revoke_jwt_refresh(config, refresh_token)
        json(conn, %{ok: true})
      end
    end
    ```
