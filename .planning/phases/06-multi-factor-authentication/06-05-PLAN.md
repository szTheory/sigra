   ```elixir
   defmodule <%= @web_module %>.MFAChallengeController do
     use <%= @web_module %>, :controller

     def new(conn, _params) do
       # Only accessible for mfa_pending sessions
       session = conn.private[:sigra_session]

       cond do
         is_nil(session) ->
           conn |> redirect(to: ~p"/users/log_in") |> halt()

         session.type != :mfa_pending ->
           # Already verified or not enrolled -- redirect appropriately (D-35)
           if <%= @app_module %>.Auth.mfa_enabled?(conn.assigns.current_scope) do
             conn |> redirect(to: ~p"/") |> halt()
           else
             conn |> redirect(to: ~p"/users/settings") |> halt()
           end

         true ->
           user = conn.assigns.current_scope
           masked_email = mask_email(user.email)
           render(conn, :mfa_challenge, masked_email: masked_email, tab: "totp")
       end
     end

     def create(conn, %{"mfa" => mfa_params}) do
       user = conn.assigns.current_scope
       code = mfa_params["code"]
       trust = mfa_params["trust"] == "true"
       method = if mfa_params["method"] == "backup", do: :backup, else: :totp

       result =
         case method do
           :totp -> <%= @app_module %>.Auth.mfa_verify(user, code)
           :backup -> <%= @app_module %>.Auth.mfa_verify_backup(user, code)
         end

       case result do
         {:ok, _} ->
           old_session = conn.private[:sigra_session]
           remember_me = get_session(conn, :mfa_remember_me) || false

           {:ok, %{session: new_session}} =
             <%= @app_module %>.Auth.complete_mfa_verification(user, old_session,
               remember_me: remember_me, trust_browser: trust)

           conn
           |> put_session(:user_token, new_session.token)
           |> delete_session(:mfa_remember_me)
           |> maybe_set_trust_cookie(user, trust)
           |> redirect(to: get_session(conn, :mfa_return_to) || ~p"/")

         {:error, :invalid_code, remaining} ->
           masked_email = mask_email(user.email)
           conn
           |> put_flash(:error, "Invalid verification code. #{remaining} attempts remaining.")
           |> render(:mfa_challenge, masked_email: masked_email, tab: mfa_params["method"] || "totp")

         {:error, :lockout, seconds} ->
           minutes = div(seconds + 59, 60)
           conn
           |> put_flash(:error, "Too many failed attempts. Try again in #{minutes} minutes.")
           |> render(:mfa_challenge, masked_email: mask_email(user.email), tab: "totp")
       end
     end

     defp mask_email(email) do
       [local, domain] = String.split(email, "@")
       first = String.first(local)
       "#{first}***@#{domain}"
     end

     defp maybe_set_trust_cookie(conn, user, true) do
       # Set trust cookie via Sigra.MFA.Trust
       cookie = Sigra.MFA.Trust.sign(
         conn.secret_key_base, user.id, user.mfa_trust_epoch,
         conn.assigns[:sigra_config].mfa.trust_ttl)
       Plug.Conn.put_resp_cookie(conn, "_sigra_mfa_trust", cookie,
         http_only: true, secure: true, same_site: "Lax",
         max_age: conn.assigns[:sigra_config].mfa.trust_ttl)
     end
     defp maybe_set_trust_cookie(conn, _user, _), do: conn
   end
   ```

</threat_model>
