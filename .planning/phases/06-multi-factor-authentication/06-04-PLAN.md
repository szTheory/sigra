1. **priv/templates/sigra.install/migration.exs**: Add MFA tables to the base migration template. Insert after existing table definitions:

   ```elixir
   # MFA Credentials (TOTP secrets, lockout tracking)
   create table(:<%= @table_prefix %>user_mfa_credentials, primary_key: false) do
     add :id, :binary_id, primary_key: true
     add :user_id, references(:<%= @table_prefix %>users, type: :binary_id, on_delete: :delete_all), null: false
     add :type, :string, null: false
     add :encrypted_secret, :binary, null: false
     add :last_used_at, :utc_datetime_usec
     add :last_verified_step, :integer
     add :failed_attempts, :integer, default: 0, null: false
     add :locked_until, :utc_datetime_usec

     timestamps(type: :utc_datetime_usec)
   end

   # enabled_at is separate from inserted_at to support enrollment flow
   # (credential row created but not enabled until code confirmed)
   alter table(:<%= @table_prefix %>user_mfa_credentials) do
     add :enabled_at, :utc_datetime_usec
   end

   create unique_index(:<%= @table_prefix %>user_mfa_credentials, [:user_id, :type])

   # Backup Codes (one row per code, atomic consumption)
   create table(:<%= @table_prefix %>user_backup_codes, primary_key: false) do
     add :id, :binary_id, primary_key: true
     add :user_id, references(:<%= @table_prefix %>users, type: :binary_id, on_delete: :delete_all), null: false
     add :hashed_code, :string, null: false
     add :used_at, :utc_datetime_usec

     timestamps(type: :utc_datetime_usec, updated_at: false)
   end

   create index(:<%= @table_prefix %>user_backup_codes, [:user_id])

   # Trust epoch on users table for mass trust cookie revocation
   alter table(:<%= @table_prefix %>users) do
     add :mfa_trust_epoch, :integer, default: 0, null: false
   end
   ```
   Per D-73, D-74, D-75, D-76, D-77, D-78.

   Note: `enabled_at` in separate `alter` because it may be nil during enrollment flow (credential created when user first submits confirmation code, but enabled_at set only after backup codes acknowledged). Actually, per D-03, secret is held in session until confirmed. So the credential is only created after confirmation. Simplify: include `enabled_at` in the create table directly.

   ```elixir
   defmodule <%= @app_module %>.Auth.UserMFACredential do
     use Ecto.Schema
     import Ecto.Changeset

     @primary_key {:id, :binary_id, autogenerate: true}
     @foreign_key_type :binary_id

     schema "<%= @table_prefix %>user_mfa_credentials" do
       belongs_to :user, <%= @app_module %>.Auth.User
       field :type, :string
       field :encrypted_secret, <%= @app_module %>.Encrypted.Binary
       field :last_used_at, :utc_datetime_usec
       field :last_verified_step, :integer
       field :failed_attempts, :integer, default: 0
       field :locked_until, :utc_datetime_usec
       field :enabled_at, :utc_datetime_usec

       timestamps(type: :utc_datetime_usec)
     end

     @doc "Changeset for creating a new MFA credential."
     def create_changeset(credential \\\\ %__MODULE__{}, attrs) do
       credential
       |> cast(attrs, [:user_id, :type, :encrypted_secret, :enabled_at])
       |> validate_required([:user_id, :type, :encrypted_secret])
       |> unique_constraint([:user_id, :type])
       |> foreign_key_constraint(:user_id)
     end
   end
   ```
   Uses `<%= @app_module %>.Encrypted.Binary` (cloak_ecto type from Phase 5 Vault setup). Per D-09, D-73, D-80.
