<objective>
Add a Playwright-driven browser smoke harness under `test/example/priv/playwright/` that exercises the full user lifecycle (register → confirm → login → sudo → session list → MFA enroll → MFA challenge → logout), plus the CI job that runs it. The harness catches the entire B1–B9 class of bugs at the browser level — things curl and ConnTest can't see (LiveSocket connect, JS bundle, form submit reaching the server).

Purpose: HTTP smoke (plan 06) catches 5xx. ConnTest catches controller/model bugs. Neither of those caught B1 (no JS bundle), B2 (no LiveSocket), B4 (live_session wiring), or B9 (form submit swallowed) in UAT — those are browser-level bugs. Playwright is the signal layer that would have caught all four. This plan also gets Phase 6 MFA partial-UAT coverage for free (the TOTP enroll + challenge flow).

This plan explicitly depends on ALL repair plans (02–05) because the golden path exercises every repair site. If any repair is still broken, the golden path fails at that step.

**fixtures/mailbox.ts:**
```typescript
// Helper to scrape the Swoosh dev mailbox for the most-recent confirmation link.
// /dev/mailbox renders emails in an iframe — use Playwright's frameLocator.

import { Page, expect } from '@playwright/test';

/**
 * Open /dev/mailbox, click the most-recent email to `recipient`, and return
 * the href of the first confirmation link inside the email body iframe.
 */
export async function extractConfirmationLink(page: Page, recipient: string): Promise<string> {
  await page.goto('/dev/mailbox');

  // The mailbox lists emails with the recipient visible as text.
  const emailLink = page.getByText(recipient).first();
  await expect(emailLink).toBeVisible({ timeout: 5_000 });
  await emailLink.click();

  // Swoosh renders the email body in an iframe.
  const href = await page
    .frameLocator('iframe')
    .locator('a[href*="/users/confirm/"]')
    .first()
    .getAttribute('href');

  if (!href) {
    throw new Error(`No confirmation link found in email for ${recipient}`);
  }

  return href;
}
```

  ```typescript
  import { test, expect } from '@playwright/test';
  import { authenticator } from 'otplib';
  import { extractConfirmationLink } from '../fixtures/mailbox';

  test('full user lifecycle: register → confirm → login → sudo → sessions → MFA enroll → MFA challenge → logout', async ({ page }) => {
    const email = `lifecycle-${Date.now()}@example.test`;
    const password = 'CorrectHorseBatteryStaple123!';

    // --- 1. Register ---
    await page.goto('/users/register');
    await page.fill('input[name="user[email]"]', email);
    await page.fill('input[name="user[password]"]', password);
    await page.click('button:has-text("Create an account")');
    // After successful register, the form trigger_submit reposts to /users/log_in?_action=registered
    // which logs the user in. Expect to land somewhere non-register.
    await expect(page).not.toHaveURL(/\/users\/register/);

    // --- 2. Confirm via mailbox ---
    const confirmHref = await extractConfirmationLink(page, email);
    await page.goto(confirmHref);
    // Confirmation LiveView auto-confirms via the token in the URL.
    await expect(page.getByText(/confirmed|confirmation/i).first()).toBeVisible({ timeout: 5_000 });

    // --- 3. Login (if not already authed) ---
    // If the registered-and-auto-logged-in flow didn't leave us signed in, hit the login form.
    await page.goto('/users/log_in');
    // The login page is now a plain controller (post plan 04). If we're already logged in,
    // the scope :redirect_if_user_is_authenticated sends us away from /users/log_in.
    if (page.url().includes('/users/log_in')) {
      await page.fill('#login_form input[name="user[email]"]', email);
      await page.fill('#login_form input[name="user[password]"]', password);
      await page.click('#login_form button:has-text("Log in")');
      await expect(page).not.toHaveURL(/\/users\/log_in(\?|$)/);
    }

    // --- 4. Sessions list (proves B6 unification: login wrote to user_sessions) ---
    await page.goto('/users/sessions');
    // The session list renders "Active just now" or similar for the current session.
    await expect(page.getByText(/active|just now|current/i).first()).toBeVisible({ timeout: 5_000 });

    // --- 5. Sudo re-auth (proves B7 sudo template fix) ---
    await page.goto('/users/sudo');
    // Expect a password form; enter the password and submit.
    await page.fill('input[name="sudo[password]"]', password);
    await page.click('button:has-text("Confirm password")');
    await expect(page).not.toHaveURL(/\/users\/sudo(\?|$)/);

    // --- 6. MFA enroll (TOTP) ---
    await page.goto('/users/settings/mfa');
    // Begin enrollment — click the enable/begin button. Exact label may vary; use the form id if present.
    const beginSelector = 'button:has-text("Enable"), button:has-text("Begin"), button:has-text("Set up")';
    await page.locator(beginSelector).first().click();

    // Read the TOTP base32 secret from the data-testid hook (Task 1).
    const secret = await page.locator('[data-testid="mfa-totp-secret"]').innerText();
    expect(secret).toBeTruthy();

    // Generate the current 6-digit code.
    const code = authenticator.generate(secret.replace(/\s+/g, ''));

    // Enter the code in the enrollment confirmation form.
    await page.fill('input[name="enroll[code]"]', code);
    await page.click('#mfa_enroll_form button:has-text("Confirm"), #mfa_enroll_form button:has-text("Verify"), #mfa_enroll_form button[type="submit"]');

    // Expect an "MFA enabled" or equivalent confirmation message.
    await expect(page.getByText(/mfa.*enabled|enrolled|enabled successfully/i).first()).toBeVisible({ timeout: 5_000 });

    // --- 7. Logout ---
    // Log out via the log_out route (DELETE /users/log_out).
    // Playwright can't issue DELETE directly from a link click easily — use page.request or click a Log out button.
    // Best: click a "Log out" button if present; fallback to page.request.delete.
    const logoutButton = page.getByRole('button', { name: /log out/i }).first();
    if (await logoutButton.count() > 0) {
      await logoutButton.click();
    } else {
      // Fallback: form-based logout (the example app may render it as a <.form method="delete">).
      await page.goto('/users/log_out');  // if router has a GET logout handler; otherwise:
      // Or issue via page.request — preserves cookies:
      // await page.request.delete('/users/log_out');
    }

    // --- 8. MFA challenge on next login ---
    await page.goto('/users/log_in');
    await page.fill('#login_form input[name="user[email]"]', email);
    await page.fill('#login_form input[name="user[password]"]', password);
    await page.click('#login_form button:has-text("Log in")');

    // Should land on the MFA challenge page because the user now has TOTP enabled.
    await expect(page).toHaveURL(/\/users\/mfa/);

    const challengeCode = authenticator.generate(secret.replace(/\s+/g, ''));
    await page.fill('input[name="mfa[code]"], input[name="challenge[code]"], input[name="totp[code]"]', challengeCode);
    await page.locator('button[type="submit"]:visible').first().click();

    // Finally logged in fully.
    await expect(page).not.toHaveURL(/\/users\/(log_in|mfa)/);
  });
  ```
