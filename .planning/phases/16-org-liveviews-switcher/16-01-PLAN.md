    defmodule Sigra.Plug.LoadOrganizationFromSlug do
      @moduledoc """
      URL-driven active organization loader (Phase 16 D-03, D-04, D-13).

      Reads `conn.params[scope_param]` (default `"org"`), resolves the
      organization by slug, verifies membership, follows 7-day slug aliases,
      and refreshes the session pointer when the URL org differs from the
      session pointer — delegating the write to
      `Sigra.Plug.put_active_organization/2` (Phase 14 D-16 single-writer).

      Returns 404 for unknown slugs AND for known slugs the current user
      is not a member of (enumeration prevention, D-04).
      """
      @behaviour Plug
      import Plug.Conn
      alias Sigra.Organizations

      @impl true
      def init(opts), do: Keyword.put_new(opts, :scope_param, "org")

      @impl true
      def call(conn, opts) do
        scope_param = Keyword.fetch!(opts, :scope_param)
        slug = conn.params[scope_param]
        config = conn.assigns.sigra_org_config || raise "sigra_org_config missing from conn.assigns"
        scope = conn.assigns.current_scope || raise "current_scope missing — LoadOrganizationFromSlug must run after fetch_current_scope"

        case resolve(config, scope, slug) do
          {:ok, org, membership} ->
            conn
            |> assign_scope(org, membership)
            |> maybe_refresh_session_pointer(scope, org, membership)

          {:redirect, new_slug} ->
            conn
            |> Phoenix.Controller.redirect(to: String.replace(conn.request_path, slug, new_slug))
            |> halt()

          :not_found ->
            conn
            |> Sigra.Plug.ErrorHandler.call(:not_found)
            |> halt()
        end
      end

      defp resolve(config, scope, slug) do
        # 1. Try live organization
        with org when not is_nil(org) <- Organizations.get_organization_by_slug(config, slug),
             membership when not is_nil(membership) <-
               Organizations.get_membership(config, scope.user, org) do
          {:ok, org, membership}
        else
          _ -> resolve_alias(config, slug)
        end
      end

      defp resolve_alias(config, slug) do
        case Organizations.get_active_slug_alias(config, slug) do
          nil -> :not_found
          alias_row ->
            current_org = Organizations.get_organization!(config, alias_row.organization_id)
            {:redirect, current_org.slug}
        end
      end

      defp assign_scope(conn, org, membership) do
        scope = conn.assigns.current_scope
        new_scope = scope.__struct__.put_active_organization(scope, org, membership)
        assign(conn, :current_scope, new_scope)
      end

      defp maybe_refresh_session_pointer(conn, old_scope, url_org, url_membership) do
        current_id = old_scope.active_organization && old_scope.active_organization.id
        if current_id != url_org.id do
          Sigra.Plug.put_active_organization(conn, url_org)
        else
          conn
        end
      end
    end
