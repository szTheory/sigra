    def list_organizations_for_user(config, user) do
      membership_schema = config.schemas.membership
      org_schema = config.schemas.organization

      from(m in membership_schema,
        where: m.user_id == ^user.id,
        join: o in ^org_schema, on: o.id == m.organization_id and is_nil(o.deleted_at),
        order_by: [desc: m.inserted_at],
        select: {o, m.role})
      |> config.repo.all()
    end

    defmodule <%= @web_namespace %>.OrganizationsLive.Index do
      use <%= @web_namespace %>, :live_view

      alias <%= @app_namespace %>.Organizations

      @impl true
      def mount(_params, _session, socket) do
        user = socket.assigns.current_scope.user
        memberships = Organizations.list_organizations_for_user(user)
        pending = Organizations.list_pending_invitations_for_user(user)

        socket =
          socket
          |> assign(:memberships, memberships)
          |> assign(:pending_invitations, pending)
          |> assign(:form, to_form(%{"name" => ""}, as: :organization))
          |> assign(:slug_preview, "")

        {:ok, socket}
      end

      @impl true
      def handle_event("validate", %{"organization" => %{"name" => name}}, socket) do
        {:noreply, assign(socket, :slug_preview, Sigra.Organizations.Slug.generate(name))}
      end

      @impl true
      def handle_event("create", %{"organization" => %{"name" => name}}, socket) do
        scope = socket.assigns.current_scope
        case Organizations.create_organization(scope, %{name: name}) do
          {:ok, org} ->
            {:noreply,
             socket
             |> put_flash(:info, "Organization created.")
             |> redirect(to: ~p"/organizations/#{org}/members")}

          {:error, changeset} ->
            {:noreply,
             socket
             |> assign(:form, to_form(changeset, as: :organization))
             |> put_flash(:error, error_flash_for(changeset))}
        end
      end

      @impl true
      def render(assigns) do
        ~H"""
        <.flash_group flash={@flash} />
        <%= cond do %>
          <% @memberships == [] and @pending_invitations == [] -> %>
            <%= render_branch_a(assigns) %>
          <% @memberships == [] and @pending_invitations != [] -> %>
            <%= render_branch_b(assigns) %>
          <% true -> %>
            <%= render_branch_c(assigns) %>
        <% end %>
        """
      end
      # ... branch functions with HEEx matching UI-SPEC verbatim
    end
