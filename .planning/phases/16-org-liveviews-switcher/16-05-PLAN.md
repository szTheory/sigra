    defmodule <%= @web_namespace %>.OrganizationMembersLive do
      use <%= @web_namespace %>, :live_view
      alias <%= @app_namespace %>.Organizations

      @page_size 100

      @impl true
      def mount(_params, _session, socket) do
        scope = socket.assigns.current_scope
        members = Organizations.list_members_with_activity(scope, limit: @page_size, offset: 0)
        total = Organizations.count_members(scope)

        socket =
          socket
          |> stream(:members, members_with_dom_id(members))
          |> assign(:total_count, total)
          |> assign(:offset, length(members))
          |> assign(:has_more, length(members) < total)
          |> assign(:pending_action, nil)
          |> assign(:role_modal_error, nil)
          |> assign(:remove_modal_error, nil)

        {:ok, socket}
      end

      defp members_with_dom_id(list) do
        Enum.map(list, fn {m, last_active} ->
          Map.put(m, :__dom_id__, "member-#{m.id}")
          |> Map.put(:__last_active__, last_active)
        end)
      end

      def handle_event("load_more", _, socket) do
        scope = socket.assigns.current_scope
        next = Organizations.list_members_with_activity(scope, limit: @page_size, offset: socket.assigns.offset)
        socket =
          Enum.reduce(members_with_dom_id(next), socket, fn m, acc ->
            stream_insert(acc, :members, m, at: -1)
          end)
          |> assign(:offset, socket.assigns.offset + length(next))
          |> assign(:has_more, socket.assigns.offset + length(next) < socket.assigns.total_count)
        {:noreply, socket}
      end

      def handle_event("open_role_modal", %{"id" => id}, socket) do
        member = find_member!(socket, id)
        {:noreply,
          socket
          |> assign(:pending_action, {:role, member})
          |> assign(:role_modal_error, nil)
          |> push_event("open-modal", %{id: "confirm-role-modal"})}
      end

      def handle_event("open_remove_modal", %{"id" => id}, socket) do
        member = find_member!(socket, id)
        {:noreply,
          socket
          |> assign(:pending_action, {:remove, member})
          |> assign(:remove_modal_error, nil)
          |> push_event("open-modal", %{id: "confirm-remove-modal"})}
      end

      def handle_event("cancel_action", _, socket) do
        {:noreply,
         socket
         |> assign(:pending_action, nil)
         |> push_event("close-modal", %{id: "confirm-role-modal"})
         |> push_event("close-modal", %{id: "confirm-remove-modal"})}
      end

      def handle_event("change_role", %{"role" => role_str}, socket) do
        {:role, membership} = socket.assigns.pending_action
        new_role = String.to_existing_atom(role_str)
        scope = socket.assigns.current_scope

        case Organizations.change_member_role(scope, membership, new_role) do
          {:ok, updated} ->
            {:noreply,
             socket
             |> stream_insert(:members, members_with_dom_id([{updated, membership.__last_active__}]) |> hd())
             |> assign(:pending_action, nil)
             |> put_flash(:info, "Role updated.")
             |> push_event("close-modal", %{id: "confirm-role-modal"})}

          {:error, :last_owner} ->
            {:noreply, assign(socket, :role_modal_error, "Cannot demote the last owner. Promote another member to owner first.")}
        end
      end

      def handle_event("remove_member", _, socket) do
        {:remove, membership} = socket.assigns.pending_action
        scope = socket.assigns.current_scope
        org_name = scope.active_organization.name
        email = membership.user.email

        case Organizations.remove_member(scope, membership) do
          {:ok, _removed} ->
            {:noreply,
             socket
             |> stream_delete(:members, membership)
             |> assign(:pending_action, nil)
             |> assign(:total_count, socket.assigns.total_count - 1)
             |> put_flash(:info, "Removed #{email} from #{org_name}.")
             |> push_event("close-modal", %{id: "confirm-remove-modal"})}

          {:error, :last_owner} ->
            {:noreply, assign(socket, :remove_modal_error, "Cannot remove the last owner. Promote another member to owner first.")}
        end
      end

      @impl true
      def render(assigns), do: ~H"""
        <.header>
          Members (<%= @total_count %>)
          <:actions>
            <.button disabled aria-disabled="true" title="Available in the next release">
              Invite member
            </.button>
          </:actions>
        </.header>

        <section id="members-section" class="overflow-x-auto">
          <.table id="members-table" rows={@streams.members}>
            <:col :let={{_id, m}} label="Email"><%= m.user.email %></:col>
            <:col :let={{_id, m}} label="Role">
              <span class={["badge", role_badge_class(m.role)]}><%= humanize_role(m.role) %></span>
            </:col>
            <:col :let={{_id, _m}} label="Status">
              <span class="badge badge-ghost">Active</span>
            </:col>
            <:col :let={{_id, m}} label="Joined"><%= relative_time(m.inserted_at) %></:col>
            <:col :let={{_id, m}} label="Last active">
              <%= if m.__last_active__, do: relative_time(m.__last_active__), else: "Never" %>
            </:col>
            <:action :let={{_id, m}}>
              <details class="dropdown dropdown-end">
                <summary class="btn btn-ghost btn-xs">
                  <.icon name="hero-ellipsis-horizontal" class="w-4 h-4" />
                </summary>
                <ul class="menu dropdown-content bg-base-100 rounded-box w-40 p-1 shadow z-10">
                  <li><button phx-click="open_role_modal" phx-value-id={m.id}>Change role</button></li>
                  <li><button class="text-error" phx-click="open_remove_modal" phx-value-id={m.id}>Remove</button></li>
                </ul>
              </details>
            </:action>
          </.table>
          <.button :if={@has_more} phx-click="load_more">Load more</.button>
        </section>

        <section id="pending-invitations-section" class="mt-8">
          <h2 class="text-lg font-semibold">Pending invitations</h2>
          <%!-- Phase 17 fills this section --%>
          <div class="card bg-base-200 p-6 text-center text-sm text-base-content/70">
            No pending invitations. Inviting members is coming in the next release.
          </div>
        </section>

        <dialog id="confirm-role-modal" class="modal" phx-hook="DialogModal">
          <%= if match?({:role, _}, @pending_action) do %>
            <% {:role, m} = @pending_action %>
            <div class="modal-box">
              <h3 class="text-lg font-semibold">Change <%= m.user.email %>'s role?</h3>
              <form phx-submit="change_role">
                <select name="role">
                  <option value="owner">Owner</option>
                  <option value="admin">Admin</option>
                  <option value="member">Member</option>
                </select>
                <%= if @role_modal_error do %>
                  <p class="text-error text-sm mt-2" role="alert"><%= @role_modal_error %></p>
                <% end %>
                <div class="modal-action">
                  <button type="submit" class="btn btn-primary">Change role</button>
                  <button type="button" class="btn btn-ghost" phx-click="cancel_action">Cancel</button>
                </div>
              </form>
            </div>
            <form method="dialog" class="modal-backdrop"><button>close</button></form>
          <% end %>
        </dialog>

        <dialog id="confirm-remove-modal" class="modal" phx-hook="DialogModal">
          <%= if match?({:remove, _}, @pending_action) do %>
            <% {:remove, m} = @pending_action %>
            <div class="modal-box">
              <h3 class="text-lg font-semibold">Remove <%= m.user.email %> from <%= @current_scope.active_organization.name %>?</h3>
              <p class="py-2"><%= m.user.email %> will be signed out of this organization immediately. You can re-invite them later.</p>
              <%= if @remove_modal_error do %>
                <p class="text-error text-sm mt-2" role="alert"><%= @remove_modal_error %></p>
              <% end %>
              <form phx-submit="remove_member">
                <div class="modal-action">
                  <button type="submit" class="btn btn-error">Remove member</button>
                  <button type="button" class="btn btn-ghost" phx-click="cancel_action">Cancel</button>
                </div>
              </form>
            </div>
            <form method="dialog" class="modal-backdrop"><button>close</button></form>
          <% end %>
        </dialog>
        """
    end
