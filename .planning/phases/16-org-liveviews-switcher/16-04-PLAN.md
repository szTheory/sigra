    defmodule <%= @web_namespace %>.OrganizationSettingsLive do
      use <%= @web_namespace %>, :live_view
      alias <%= @app_namespace %>.Organizations

      @impl true
      def mount(_params, _session, socket) do
        scope = socket.assigns.current_scope
        org = scope.active_organization

        socket =
          socket
          |> assign(:org, org)
          |> assign(:rename_form, to_form(%{"name" => org.name}, as: :organization))
          |> assign(:slug_form_open?, false)
          |> assign(:delete_form_open?, false)
          |> assign(:slug_form, to_form(%{"slug" => "", "password" => "", "confirm_slug" => ""}, as: :slug_change))
          |> assign(:delete_form, to_form(%{"password" => "", "confirm_name" => ""}, as: :delete_org))

        {:ok, socket}
      end

      def handle_event("rename", %{"organization" => params}, socket) do
        case Organizations.rename_organization(socket.assigns.current_scope, params) do
          {:ok, org} ->
            {:noreply,
             socket
             |> assign(:org, org)
             |> assign(:rename_form, to_form(%{"name" => org.name}, as: :organization))
             |> put_flash(:info, "Name updated.")}

          {:error, changeset} ->
            {:noreply, assign(socket, :rename_form, to_form(changeset, as: :organization))}
        end
      end

      def handle_event("open_slug_form", _, socket),
        do: {:noreply, assign(socket, :slug_form_open?, true)}
      def handle_event("close_slug_form", _, socket),
        do: {:noreply, assign(socket, :slug_form_open?, false)}
      def handle_event("open_delete_form", _, socket),
        do: {:noreply, assign(socket, :delete_form_open?, true)}
      def handle_event("close_delete_form", _, socket),
        do: {:noreply, assign(socket, :delete_form_open?, false)}

      def handle_event("update_slug", %{"slug_change" => params}, socket) do
        case Organizations.update_slug(socket.assigns.current_scope, params) do
          {:ok, org} ->
            {:noreply,
             socket
             |> put_flash(:info, "Slug updated. The old slug redirects for 7 days.")
             |> redirect(to: ~p"/organizations/#{org}/settings")}

          {:error, :invalid_password} ->
            changeset =
              Ecto.Changeset.change({params, %{password: :string, slug: :string, confirm_slug: :string}})
              |> Ecto.Changeset.add_error(:password, "That password is incorrect.")
            {:noreply, assign(socket, :slug_form, to_form(changeset, as: :slug_change))}

          {:error, %Ecto.Changeset{} = changeset} ->
            # Map library changeset errors to UI-SPEC copy:
            changeset = remap_slug_errors(changeset)
            {:noreply, assign(socket, :slug_form, to_form(changeset, as: :slug_change))}
        end
      end

      def handle_event("soft_delete", %{"delete_org" => params}, socket) do
        case Organizations.soft_delete_organization(socket.assigns.current_scope, params) do
          {:ok, _org} ->
            {:noreply,
             socket
             |> put_flash(:info, "Organization deleted.")
             |> redirect(to: ~p"/organizations")}

          {:error, :invalid_password} ->
            changeset =
              Ecto.Changeset.change({params, %{password: :string, confirm_name: :string}})
              |> Ecto.Changeset.add_error(:password, "That password is incorrect.")
            {:noreply, assign(socket, :delete_form, to_form(changeset, as: :delete_org))}

          {:error, %Ecto.Changeset{} = changeset} ->
            changeset = remap_delete_errors(changeset, socket.assigns.org.name)
            {:noreply, assign(socket, :delete_form, to_form(changeset, as: :delete_org))}
        end
      end

      end

        </.header>

        <section class="bg-base-200 p-6 rounded-lg mt-6">
          <h2 class="text-lg font-semibold">General</h2>
          <.form for={@rename_form} phx-submit="rename">
            <.input field={@rename_form[:name]} label="Organization name" />
            <.button type="submit" phx-disable-with="Saving...">Save name</.button>
          </.form>
        </section>

        <section class="bg-base-200 p-6 rounded-lg mt-8">
          <h2 class="text-lg font-semibold">Slug</h2>
          <p class="text-sm text-base-content/70">Current: <code><%= @org.slug %></code></p>
          <%= if @slug_form_open? do %>
            <.form for={@slug_form} phx-submit="update_slug">
              <.input field={@slug_form[:slug]} label="New slug" />
              <.input field={@slug_form[:password]} type="password" label="Current password" />
              <.input field={@slug_form[:confirm_slug]} label={"Type " <> @org.slug <> " to confirm"} />
              <div role="alert" class="alert alert-warning alert-soft">
                <.icon name="hero-exclamation-triangle" class="w-5 h-5" />
                Your current slug <code><%= @org.slug %></code> will redirect to the new slug for 7 days, after which it becomes available to other organizations. Links and bookmarks using <code><%= @org.slug %></code> will continue to work during that window.
              </div>
              <.button type="submit" class="btn btn-error" phx-disable-with="Updating...">Update slug</.button>
              <.button type="button" phx-click="close_slug_form" class="btn btn-ghost">Cancel</.button>
            </.form>
          <% else %>
            <.button phx-click="open_slug_form">Change slug</.button>
