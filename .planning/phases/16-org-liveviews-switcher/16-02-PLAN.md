    ```elixir
    defmodule <%= @web_namespace %>.OrganizationSwitchController do
      use <%= @web_namespace %>, :controller

      alias <%= @app_namespace %>.Organizations

      @doc """
      POST /organizations/switch — sensitive mutation via POST (D-05 / ORG-UX-03).

      Body: %{"organization_id" => id, "return_to" => local_path}
      """
      def update(conn, %{"organization_id" => org_id} = params) do
        scope = conn.assigns.current_scope
        return_to = params |> Map.get("return_to", "/") |> Sigra.UrlSafety.local_path()

        with {:ok, org} <- fetch_member_org(scope, org_id) do
          conn
          |> Organizations.set_active_organization(org)
          |> put_flash(:info, "Switched to #{org.name}.")
          |> redirect(to: return_to)
        else
          _ ->
            conn
            |> put_status(:not_found)
            |> put_view(html: <%= @web_namespace %>.ErrorHTML)
            |> render(:"404")
            |> halt()
        end
      end

      defp fetch_member_org(scope, org_id) do
        case Organizations.get_membership_by_org_id(scope.user, org_id) do
          nil -> :error
          membership -> {:ok, membership.organization}
        end
      end
    end
    ```
